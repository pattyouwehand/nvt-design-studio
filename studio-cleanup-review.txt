import { NextRequest, NextResponse } from 'next/server'
import { randomUUID } from 'node:crypto'
import { prisma } from '@/lib/prisma'
import { getStudioUser } from '@/lib/auth'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'
import { getProductType } from '@/lib/catalog/products'

export const runtime = 'nodejs'

export async function POST(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  if (!await getStudioUser()) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const { id } = await params
  const brand = request.nextUrl.searchParams.get('brand')
  if (!brand || !isBrandId(brand) || !isArtworkId(id)) return NextResponse.json({ error: 'Invalid artwork.' }, { status: 400 })
  try {
    const artwork = await prisma.artwork.findFirst({ where: { id, brand }, select: { id: true } })
    if (!artwork) return NextResponse.json({ error: 'Artwork not found.' }, { status: 404 })
    const body: unknown = await request.json()
    const typeId = body && typeof body === 'object' && 'productType' in body ? body.productType : null
    const productType = typeof typeId === 'string' ? getProductType(typeId) : undefined
    if (!productType) return NextResponse.json({ error: 'Invalid product type.' }, { status: 400 })
    const productId = randomUUID()
    const product = await prisma.product.create({
      data: {
        id: productId, artworkId: id, brand, productType: productType.id,
        title: '', slug: `draft-${productId}`, status: 'draft',
        subtitle: '', description: '', price: null,
        sizes: [], colors: [], category: '', label: '', collections: []
      },
      select: { id: true }
    })
    return NextResponse.json({ success: true, productId: product.id }, { status: 201 })
  } catch (error) {
    console.error('Unable to create product:', error)
    return NextResponse.json({ error: 'Unable to create product.' }, { status: 500 })
  }
}

========== app/api/artworks/[id]/route.ts ==========

import { NextRequest, NextResponse } from 'next/server'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'
import { getStudioUser } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { deleteArtworkFiles } from '@/lib/artworkStorage'

export const runtime = 'nodejs'
type Context = { params: Promise<{ id: string }> }

async function identify(request: NextRequest, context: Context) {
  const { id } = await context.params
  const brand = request.nextUrl.searchParams.get('brand')

  return brand && isBrandId(brand) && isArtworkId(id) ? { id, brand } : null
}

export async function GET(request: NextRequest, context: Context) {
  if (!(await getStudioUser())) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const identity = await identify(request, context)
  if (!identity) return NextResponse.json({ error: 'Invalid artwork' }, { status: 400 })
  const artwork = await prisma.artwork.findFirst({ where: identity })
  if (!artwork) return NextResponse.json({ error: 'Artwork not found' }, { status: 404 })
  const [products, published] = await Promise.all([
    prisma.product.findMany({
      where: { artworkId: identity.id, brand: identity.brand },
      select: { id: true, title: true, slug: true, productType: true, status: true, createdAt: true },
      orderBy: { createdAt: 'asc' }
    }),
    prisma.publishedProduct.findMany({
      where: { brand: identity.brand }, select: { productId: true }
    })
  ])
  const publishedIds = new Set(published.map((item) => item.productId))
  return NextResponse.json({
    id: artwork.id, brand: artwork.brand, filename: artwork.filename,
    createdAt: artwork.createdAt.toISOString(),
    linkedProducts: products.map((product) => ({ ...product, isPublished: publishedIds.has(product.id) }))
  })
}

export async function PUT() {
  return NextResponse.json({ error: 'Artwork metadata editing has moved to the Product Editor.' }, { status: 410 })
}

export async function DELETE(request: NextRequest, context: Context) {
  if (!(await getStudioUser())) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const identity = await identify(request, context)
  if (!identity) return NextResponse.json({ error: 'Invalid artwork' }, { status: 400 })
  if (await prisma.product.count({ where: { artworkId: identity.id, brand: identity.brand } })) {
    return NextResponse.json({ error: 'Delete linked products first.' }, { status: 409 })
  }
  const result = await prisma.artwork.deleteMany({ where: identity })
  if (!result.count) return NextResponse.json({ error: 'Artwork not found' }, { status: 404 })
  try { await deleteArtworkFiles(identity.brand, identity.id) }
  catch (error) { console.error('Artwork deleted; storage cleanup failed:', error); return NextResponse.json({ success: true, warning: 'Storage cleanup failed.' }) }
  return NextResponse.json({ success: true })
}

import { randomUUID } from 'node:crypto'
import { NextRequest, NextResponse } from 'next/server'
import sharp from 'sharp'
import { isBrandId } from '@/lib/brands'
import { getStudioUser } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { uploadArtworkImage } from '@/lib/artworkStorage'
import { supabaseAdmin } from '@/lib/supabaseAdmin'

export const runtime = 'nodejs'

const MAX_FILE_SIZE = 25 * 1024 * 1024

export async function GET() {
  if (!(await getStudioUser())) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const artworks = await prisma.artwork.findMany({
      include: {
        _count: {
          select: { products: true }
        }
      },
      orderBy: { createdAt: 'desc' }
    })

    return NextResponse.json({
      artworks: artworks.map((artwork) => ({
        id: artwork.id,
        brand: artwork.brand,
        filename: artwork.filename,
        createdAt: artwork.createdAt.toISOString(),
        updatedAt: artwork.updatedAt.toISOString(),
        productCount: artwork._count.products,
      })),
    })
  } catch (error) {
    console.error('Unable to load artworks:', error)

    return NextResponse.json(
      { error: 'Unable to load artworks.' },
      { status: 500 }
    )
  }
}

export async function POST(request: NextRequest) {
  if (!(await getStudioUser())) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const formData = await request.formData()
    const brand = formData.get('brand')
    const file = formData.get('artwork')

    if (
      typeof brand !== 'string' ||
      !isBrandId(brand)
    ) {
      return NextResponse.json(
        { error: 'Invalid brand.' },
        { status: 400 }
      )
    }

    if (!(file instanceof File)) {
      return NextResponse.json(
        { error: 'Please select a PNG file.' },
        { status: 400 }
      )
    }

    if (
      file.size === 0 ||
      file.size > MAX_FILE_SIZE
    ) {
      return NextResponse.json(
        { error: 'File must be smaller than 25 MB.' },
        { status: 400 }
      )
    }

    const buffer = Buffer.from(
      await file.arrayBuffer()
    )

    const metadata = await sharp(buffer).metadata()

    if (
      metadata.format !== 'png' ||
      !metadata.width ||
      !metadata.height ||
      metadata.width > 10000 ||
      metadata.height > 10000
    ) {
      return NextResponse.json(
        { error: 'Please upload a valid PNG image.' },
        { status: 400 }
      )
    }

    const id = randomUUID()

    await uploadArtworkImage(brand, id, buffer)

    try {
      const artwork = await prisma.artwork.create({
        data: {
          id,
          brand,
          filename: file.name,
        },
      })

      return NextResponse.json(
        {
          success: true,
          artwork: {
            id: artwork.id,
            brand: artwork.brand,
            filename: artwork.filename,
            width: metadata.width,
            height: metadata.height
          }
        },
        { status: 201 }
      )
    } catch (databaseError) {
      const { error: cleanupError } =
        await supabaseAdmin.storage
          .from('artworks')
          .remove([`${brand}/${id}/artwork.png`])

      if (cleanupError) {
        console.error(
          'Unable to clean up uploaded artwork:',
          cleanupError
        )
      }

      throw databaseError
    }
  } catch (error) {
    console.error('Artwork upload failed:', error)

    return NextResponse.json(
      { error: 'Unable to upload artwork.' },
      { status: 500 }
    )
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'
import { getProductType } from '@/lib/catalog/products'
import { getArtworkMockup } from '@/lib/artworkStorage'
import { getStudioUser } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export const runtime = 'nodejs'
type Context = { params: Promise<{ id: string }> }
async function serve(request: NextRequest, { params }: Context, head: boolean) {
  if (!(await getStudioUser())) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const { id } = await params
  const brand = request.nextUrl.searchParams.get('brand')
  const type = request.nextUrl.searchParams.get('productType') ?? 'classic-tee'
  const color = request.nextUrl.searchParams.get('color') ?? 'black'
  const product = getProductType(type)
  if (!brand || !isBrandId(brand) || !isArtworkId(id) || !product?.colors.some((item) => item.id === color)) {
    return NextResponse.json({ error: 'Invalid mockup parameters' }, { status: 400 })
  }
  const exists = await prisma.artwork.findFirst({ where: { id, brand }, select: { id: true } })
  if (!exists) return NextResponse.json({ error: 'Artwork not found' }, { status: 404 })
  try {
    const buffer = await getArtworkMockup(brand, id, `${type}-${color}-front.webp`)
    return new NextResponse(head ? null : new Uint8Array(buffer), { status: 200, headers: { 'Content-Type': 'image/webp', 'Cache-Control': 'no-store' } })
  } catch {
    return NextResponse.json({ error: 'Mockup not found' }, { status: 404 })
  }
}
export async function GET(request: NextRequest, context: Context) { return serve(request, context, false) }
export async function HEAD(request: NextRequest, context: Context) { return serve(request, context, true) }

========== app/api/products/[id]/publish/route.ts ==========

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getStudioUser } from '@/lib/auth'
import { getProductType } from '@/lib/catalog/products'
import { getArtworkMockup } from '@/lib/artworkStorage'
import { supabaseAdmin } from '@/lib/supabaseAdmin'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'

export const runtime = 'nodejs'

const PUBLIC_BUCKET = 'product-images'

export async function POST(_request: Request, { params }: { params: Promise<{ id: string }> }) {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const { id } = await params

  if (!isArtworkId(id)) {
    return NextResponse.json(
      { error: 'Invalid product ID.' },
      { status: 400 }
    )
  }

  try {
    const product = await prisma.product.findUnique({
      where: { id },
      include: { artwork: true }
    })

    if (!product) {
      return NextResponse.json(
        { error: 'Product not found.' },
        { status: 404 }
      )
    }

    const brand = product.brand

    if (
      !isBrandId(brand) ||
      product.artwork.brand !== brand
    ) {
      return NextResponse.json(
        { error: 'Invalid product brand.' },
        { status: 400 }
      )
    }

    const productType = getProductType(product.productType)

    if (!productType) {
      return NextResponse.json(
        { error: 'Unknown product type.' },
        { status: 400 }
      )
    }

    const sizes = product.sizes
    const colors = product.colors
    const collections = product.collections

    if (
      product.status !== 'ready' ||
      !product.title.trim() ||
      !product.slug.trim() ||
      product.price === null ||
      !Number.isFinite(product.price) ||
      product.price <= 0 ||
      !Array.isArray(sizes) ||
      sizes.length === 0 ||
      !sizes.every((size) => typeof size === 'string') ||
      !Array.isArray(colors) ||
      colors.length === 0 ||
      !colors.every((color) => typeof color === 'string') ||
      !colors.every((color) =>
        productType.colors.some((item) => item.id === color)
      )
    ) {
      return NextResponse.json(
        { error: 'Complete the product details before publishing.' },
        { status: 400 }
      )
    }

    // Prevent another product from using the same brand/slug.
    const slugConflict = await prisma.publishedProduct.findFirst({
      where: {
        brand,
        slug: product.slug,
        productId: { not: product.id }
      },
      select: { id: true }
    })

    if (slugConflict) {
      return NextResponse.json(
        { error: 'This URL slug is already published for this brand.' },
        { status: 409 }
      )
    }

    const images: Record<string, string> = {}
    const uploadedPaths: string[] = []

    // Upload public mockups before updating the published snapshot.
    for (const color of colors as string[]) {
      const filename = `${productType.id}-${color}-front.webp`

      // Read from the existing private artworks bucket.
      const mockup = await getArtworkMockup(
        brand,
        product.artworkId,
        filename
      )

      // Product ID ensures different clothing types do not collide.
      const storagePath = `${brand}/${product.id}/${filename}`

      const { error: uploadError } = await supabaseAdmin.storage
        .from(PUBLIC_BUCKET)
        .upload(storagePath, mockup, {
          contentType: 'image/webp',
          upsert: true,
          cacheControl: '60'
        })

      if (uploadError) {
        throw uploadError
      }

      uploadedPaths.push(storagePath)

      const { data } = supabaseAdmin.storage
        .from(PUBLIC_BUCKET)
        .getPublicUrl(storagePath)

      // Version query helps browsers refresh updated mockups.
      images[color] = `${data.publicUrl}?v=${Date.now()}`
    }

    const snapshot = {
      id: product.id,
      artworkId: product.artworkId,
      brand,
      productType: product.productType,
      title: product.title,
      slug: product.slug,
      subtitle: product.subtitle,
      description: product.description ?? '',
      price: product.price,
      sizes,
      colors,
      category: product.category,
      collections: Array.isArray(collections) ? collections : [],
      images
    }

    const published = await prisma.publishedProduct.upsert({
      where: { productId: product.id },
      create: {
        productId: product.id,
        brand,
        slug: product.slug,
        data: snapshot
      },
      update: {
        brand,
        slug: product.slug,
        data: snapshot
      }
    })

    return NextResponse.json({
      success: true,
      productId: published.productId,
      publishedAt: published.publishedAt,
      images: uploadedPaths.length
    })
  } catch (error) {
    console.error('Product publishing failed:', error)

    return NextResponse.json(
      {

export async function PATCH( request: NextRequest, context: RouteContext ) {

export async function PUT(
  request: NextRequest,
  context: RouteContext
) {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const { id } = await context.params

    const existingImage = await prisma.sharedImage.findUnique({
      where: { id }
    })

    if (!existingImage) {
      return NextResponse.json(
        { error: 'Image not found.' },
        { status: 404 }
      )
    }

    const formData = await request.formData()
    const file = formData.get('image')

    if (!(file instanceof File)) {
      return NextResponse.json(
        { error: 'Please select an image.' },
        { status: 400 }
      )
    }

    if (
      !['image/png', 'image/jpeg', 'image/webp'].includes(file.type)
    ) {
      return NextResponse.json(
        { error: 'Only PNG, JPEG and WebP images are allowed.' },
        { status: 400 }
      )
    }

    if (file.size === 0 || file.size > MAX_FILE_SIZE) {
      return NextResponse.json(
        { error: 'Image must be smaller than 15 MB.' },
        { status: 400 }
      )
    }

    const buffer = Buffer.from(await file.arrayBuffer())

    let optimizedImage: Buffer

    try {
      optimizedImage = await sharp(buffer, {
        limitInputPixels: 40_000_000
      })
        .rotate()
        .resize(1600, 1600, {
          fit: 'inside',
          withoutEnlargement: true
        })
        .webp({
          quality: 85,
          effort: 6
        })
        .toBuffer()
    } catch {
      return NextResponse.json(
        { error: 'Invalid or unsupported image file.' },
        { status: 400 }
      )
    }

    // A unique path avoids serving an outdated cached image.
    const newStoragePath = `shared-images/${id}-${randomUUID()}.webp`

    const { error: uploadError } = await supabaseAdmin.storage
      .from(BUCKET)
      .upload(newStoragePath, optimizedImage, {
        contentType: 'image/webp',
        upsert: false
      })

    if (uploadError) {
      console.error('Unable to upload replacement:', uploadError)

      return NextResponse.json(
        { error: 'Unable to upload replacement image.' },
        { status: 500 }
      )
    }

    try {
      const image = await prisma.sharedImage.update({
        where: { id },
        data: {
          storagePath: newStoragePath
        }
      })

      // Do not delete the previous file here.
      // Published storefronts may still reference its URL.
      return NextResponse.json({
        success: true,
        image
      })
    } catch (error) {
      await supabaseAdmin.storage
        .from(BUCKET)
        .remove([newStoragePath])

      throw error
    }
  } catch (error) {
    console.error('Unable to replace shared image:', error)

    return NextResponse.json(
      { error: 'Unable to replace image.' },
      { status: 500 }
    )
  }
}


export async function DELETE(
  _request: NextRequest,
  context: RouteContext
) {
  const user = await getStudioUser()

    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

    const existingImage = await prisma.sharedImage.findUnique({
      where: { id }
    })

    if (!existingImage) {
      return NextResponse.json(
        { error: 'Image not found.' },
        { status: 404 }
      )
    }

    // Never delete an image linked to a product.
    const linkedProducts = await prisma.productSharedImage.count({
      where: {
        sharedImageId: id
      }
    })

    if (linkedProducts > 0) {
      return NextResponse.json(
        {
          error: `This image is used by ${linkedProducts} product(s). Remove the product links before deleting it.`
        },
        { status: 409 }
      )
    }

    // Delete the database record first.
    // The relation check above prevents deleting linked images.
    await prisma.sharedImage.delete({
      where: { id }
    })

    // Remove the current file from private Supabase Storage.
    const { error: storageError } = await supabaseAdmin.storage
      .from(BUCKET)
      .remove([existingImage.storagePath])

    if (storageError) {
      console.error(
        'Shared image deleted from database, but storage cleanup failed:',
        storageError
      )

      return NextResponse.json({
        success: true,
        warning: 'Image deleted, but storage cleanup could not be completed.'
      })
    }

    return NextResponse.json({
      success: true
    })
  } catch (error) {
    console.error('Unable to delete shared image:', error)

    return NextResponse.json(
      { error: 'Unable to delete image.' },
      { status: 500 }
    )
  }
}

export const runtime = 'nodejs'

const BUCKET = 'artworks'
const MAX_FILE_SIZE = 25 * 1024 * 1024
const MAX_DIMENSION = 10000

export async function POST(request: NextRequest) {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const formData = await request.formData()

    const file = formData.get('image')
    const name = formData.get('name')
    const productType = formData.get('productType')
    const color = formData.get('color')

    if (!(file instanceof File)) {
      return NextResponse.json(
        { error: 'Please select an image.' },
        { status: 400 }
      )
    }

    if (
      typeof name !== 'string' ||
      !name.trim() ||
      name.trim().length > 150
    ) {
      return NextResponse.json(
        { error: 'Please enter a valid image name.' },
        { status: 400 }
      )
    }

    if (file.size === 0 || file.size > MAX_FILE_SIZE) {
      return NextResponse.json(
        { error: 'Image must be between 1 byte and 25 MB.' },
        { status: 400 }
      )
    }

    if (
      (productType !== null && typeof productType !== 'string') ||
      (color !== null && typeof color !== 'string') ||
      (typeof productType === 'string' && productType.length > 100) ||
      (typeof color === 'string' && color.length > 100)
    ) {
      return NextResponse.json(
        { error: 'Invalid product type or color.' },
        { status: 400 }
      )
    }

    const buffer = Buffer.from(await file.arrayBuffer())
    const metadata = await sharp(buffer).metadata()

    if (
      !metadata.format ||
      !['png', 'jpeg', 'webp'].includes(metadata.format) ||
      !metadata.width ||
      !metadata.height ||
      metadata.width > MAX_DIMENSION ||
      metadata.height > MAX_DIMENSION
    ) {
      return NextResponse.json(
        { error: 'Only valid PNG, JPEG and WebP images are supported.' },
        { status: 400 }
      )
    }

    // Convert the uploaded image to an optimized WebP.
    const optimizedImage = await sharp(buffer)
      .rotate()
      .resize({
        width: 1600,
        height: 1600,
        fit: 'inside',
        withoutEnlargement: true
      })
      .webp({
        quality: 85,
        effort: 6
      })
      .toBuffer()

    const id = randomUUID()
    const storagePath = `shared-images/${id}.webp`

    // Upload the optimized image to Supabase Storage.
    const { error: uploadError } = await supabaseAdmin.storage
      .from(BUCKET)
      .upload(storagePath, optimizedImage, {
        contentType: 'image/webp',
        upsert: false
      })

    if (uploadError) {
      throw uploadError
    }

    try {
      // Save image metadata in PostgreSQL.
      const image = await prisma.sharedImage.create({
        data: {
          id,
          name: name.trim(),
          storagePath,
          productType:
            typeof productType === 'string'
              ? productType.trim() || null
              : null,
          color:
            typeof color === 'string'
              ? color.trim() || null
              : null
        }
      })

        {
          success: true,
          image
        },
        { status: 201 }
      )
    } catch (databaseError) {
      // Remove the uploaded file if saving to the database fails.
      const { error: cleanupError } = await supabaseAdmin.storage
        .from(BUCKET)
        .remove([storagePath])

      if (cleanupError) {
        console.error(
          'Unable to clean up shared image:',
          cleanupError
        )
      }

      throw databaseError
    }
  } catch (error) {
    console.error('Shared image upload failed:', error)

    )

  const [replacingId, setReplacingId] = useState<string | null>(null)
  const [replacementFile, setReplacementFile] = useState<File | null>(null)
  const [replacementPreview, setReplacementPreview] = useState<string | null>(null)
  const [deletingId, setDeletingId] = useState<string | null>(null)

  useEffect(() => {
    async function loadImages() {
      try {
        const response = await fetch('/api/shared-images', {
          cache: 'no-store'
        })

        const result = await response.json()

        if (!response.ok) {
          throw new Error(result.error || 'Unable to load images.')
        }

        setImages(result.images)
      } catch (error) {
        setError(
          error instanceof Error
            ? error.message
            : 'Unable to load images.'
        )
      } finally {
        setLoading(false)
      }
    }

    void loadImages()
  }, [])

  useEffect(() => {
    if (!replacementFile) {
      setReplacementPreview(null)
      return
    }

    const previewUrl = URL.createObjectURL(replacementFile)
    setReplacementPreview(previewUrl)

    return () => {
      URL.revokeObjectURL(previewUrl)
    }
  }, [replacementFile])

  const productTypes = useMemo(() => {
    return [...new Set(
      images
        .map((image) => image.productType)
        .filter((value): value is string => Boolean(value))
    )].sort()
  }, [images])

  const filteredImages = useMemo(() => {
    return images.filter(
      (image) =>
        filter === 'all' ||
        image.productType === filter
    )
  }, [images, filter])

  function startEditing(image: SharedImage) {
    if (replacing || saving) return

    cancelReplacing()
    setEditingId(image.id)
    setEditForm({
      name: image.name,
      productType: image.productType ?? '',
      color: image.color ?? ''
    })
    setError('')
    setMessage('')
  }

  function cancelEditing() {
    setEditingId(null)
    setEditForm({
      name: '',
      productType: '',
      color: ''
    })
  }

  async function handleSave(id: string) {
    if (!editForm.name.trim() || saving) return

    setSaving(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(`/api/shared-images/${id}`, {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          name: editForm.name.trim(),
          productType: editForm.productType.trim() || null,
          color: editForm.color.trim() || null
        })
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to update image.')
      }

      setImages((current) =>
        current.map((image) =>
          image.id === id ? result.image : image
        )
      )

      cancelEditing()
      setMessage('Image updated successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to update image.'
      )
    } finally {
      setSaving(false)
    }
  }

  function startReplacing(image: SharedImage) {
    if (saving || replacing) return

    cancelEditing()
    setReplacingId(image.id)
    setReplacementFile(null)
    setError('')
    setMessage('')
  }

  function cancelReplacing() {
    setReplacingId(null)
    setReplacementFile(null)
  }

  function handleReplacementFile(event: ChangeEvent<HTMLInputElement>) {
    const selectedFile = event.target.files?.[0] ?? null

    if (!selectedFile) {
      setReplacementFile(null)
      return
    }

    if (!ALLOWED_TYPES.includes(selectedFile.type)) {
      setError('Only PNG, JPEG and WebP images are allowed.')
      setReplacementFile(null)
      event.target.value = ''
      return
    }

    if (selectedFile.size === 0 || selectedFile.size > MAX_FILE_SIZE) {
      setError('Image must be smaller than 15 MB.')
      setReplacementFile(null)
      event.target.value = ''
      return
    }

    setError('')
    setMessage('')
    setReplacementFile(selectedFile)
  }

  async function handleReplace(id: string) {
    if (!replacementFile || replacing) return

    setReplacing(true)
    setError('')
    setMessage('')

    try {
      const formData = new FormData()
      formData.append('image', replacementFile)

      const response = await fetch(`/api/shared-images/${id}`, {
        method: 'PUT',
        body: formData
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to replace image.')
      }

      setImages((current) =>
        current.map((image) =>
          image.id === id ? result.image : image
        )
      )

      cancelReplacing()
      setMessage('Image replaced successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to replace image.'
      )
    } finally {
      setReplacing(false)
    }
  }

  async function handleDelete(image: SharedImage) {
    const confirmed = window.confirm(
      `Are you sure you want to permanently delete "${image.name}"?\n\nThis action cannot be undone.`
    )

    if (!confirmed) return

    setDeletingId(image.id)
    setError('')
    setMessage('')

    try {
      const response = await fetch(`/api/shared-images/${image.id}`, {
        method: 'DELETE'
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to delete image.')
      }

      setImages((current) =>
        current.filter((item) => item.id !== image.id)
      )

      setMessage(
        result.warning || 'Image deleted successfully!'
      )
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to delete image.'
      )
    } finally {
      setDeletingId(null)
    }
  }

  async function handleUpload(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()

    if (!file || !name.trim()) return

    setUploading(true)
    setError('')
    setMessage('')

    try {
      const formData = new FormData()
      formData.append('image', file)
      formData.append('name', name.trim())
      formData.append('productType', productType.trim())
      formData.append('color', color.trim())

      const response = await fetch('/api/shared-images', {
        method: 'POST',
        body: formData
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to upload image.')
      }

      setImages((current) => [result.image, ...current])

      setFile(null)
      setName('')
      setProductType('')
      setColor('')
      setMessage('Image uploaded successfully!')

      const fileInput = document.getElementById(
        'shared-image-file'
      ) as HTMLInputElement | null

      if (fileInput) fileInput.value = ''
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to upload image.'
      )
    } finally {
      setUploading(false)
    }
  }

  return (
    <main className="min-h-screen bg-[#F5F2EB] px-6 py-12 text-[#171717]">
      <div className="mx-auto max-w-7xl">
        <header className="mb-10 flex flex-wrap items-end justify-between gap-6">
          <div>
            <p className="mb-2 text-sm font-medium uppercase tracking-widest text-neutral-500">
              NVT Design Studio
            </p>

            <h1 className="text-4xl font-bold">
              Shared Image Library
            </h1>

            <p className="mt-3 text-neutral-600">
              Manage reusable product photos for your storefronts.
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-3">
            <Link
              href="/library"
              className="rounded-lg border border-neutral-300 bg-white px-5 py-3 text-sm font-semibold transition hover:bg-neutral-100"
            >
              Artwork Library
            </Link>
          </div>
        </header>

        {/* Overview */}
        <div className="mb-8 grid gap-4 sm:grid-cols-3">
          <div className="rounded-xl border border-neutral-200 bg-white p-5">
            <p className="text-sm text-neutral-500">
              Total images
            </p>
            <p className="mt-2 text-3xl font-bold">
              {images.length}
            </p>
          </div>

          <div className="rounded-xl border border-neutral-200 bg-white p-5">
            <p className="text-sm text-neutral-500">
              Product types
            </p>
            <p className="mt-2 text-3xl font-bold">
              {productTypes.length}
            </p>
          </div>

          <div className="rounded-xl border border-neutral-200 bg-white p-5">
            <p className="text-sm text-neutral-500">
              Showing
            </p>
            <p className="mt-2 text-3xl font-bold">
              {filteredImages.length}
            </p>
          </div>
        </div>

        {/* Upload */}
        <section className="mb-10 rounded-xl border border-neutral-200 bg-white p-6">
          <h2 className="text-xl font-bold">
            Upload shared image
          </h2>

          <p className="mt-2 text-sm text-neutral-500">
            Upload PNG, JPEG or WebP. Images are automatically optimized to WebP.
          </p>

          <form
            onSubmit={handleUpload}
            className="mt-6 grid gap-4 md:grid-cols-2"
          >
            <label className="block text-sm font-medium">
              Image
              <input
                id="shared-image-file"
                type="file"
                accept="image/png,image/jpeg,image/webp"
                required
                onChange={(event) => {
                  const selectedFile = event.target.files?.[0] ?? null
                  setFile(selectedFile)

                  if (selectedFile && !name) {
                    setName(selectedFile.name.replace(/\.[^.]+$/, ''))
                  }
                }}
                className="mt-2 block w-full rounded-lg border border-neutral-300 p-3 text-sm"
              />
            </label>

            <label className="block text-sm font-medium">
              Image name
              <input
                type="text"
                value={name}
                onChange={(event) => setName(event.target.value)}
                maxLength={150}
                required
                placeholder="Black T-shirt back"
                className="mt-2 w-full rounded-lg border border-neutral-300 px-4 py-3 text-sm"
              />
            </label>

            <label className="block text-sm font-medium">
              Product type (optional)
              <input
                type="text"
                value={productType}
                onChange={(event) => setProductType(event.target.value)}
                maxLength={100}
                placeholder="t-shirt"
                className="mt-2 w-full rounded-lg border border-neutral-300 px-4 py-3 text-sm"
              />

            <label className="block text-sm font-medium">
              Color (optional)

                type="text"
                value={color}
                onChange={(event) => setColor(event.target.value)}
                maxLength={100}
                placeholder="black"

            Your images
          </h2>

          <select

            onChange={(event) => setFilter(event.target.value)}

            <option value="all">All product types</option>

            {productTypes.map((type) => (

                {type}
              </option>

        </div>

        {loading && (

            Loading images...
          </p>

        {!loading && filteredImages.length === 0 && (

            <h3 className="text-lg font-semibold">
              No images found

            <p className="mt-2 text-neutral-500">
              Upload your first shared product image.
            </p>
          </div>

        {/* Image grid */}

                        Image name

                            setEditForm((current) => ({
                              ...current,
                              name: event.target.value

                            />
                          </div>
                        </div>
                      )}

                      <div className="flex flex-wrap gap-2">
                        <button
                          type="button"
                          onClick={() => void handleReplace(image.id)}
                          disabled={!replacementFile || replacing}
                          className="rounded-lg bg-neutral-900 px-4 py-2 text-sm font-semibold text-white transition hover:bg-neutral-700 disabled:cursor-not-allowed disabled:opacity-50"
                        >

                        <button
                          type="button"
                          onClick={cancelReplacing}
                          disabled={replacing}
                          className="rounded-lg border border-neutral-300 px-4 py-2 text-sm font-semibold transition hover:bg-neutral-100 disabled:opacity-50"
                        >
                          Cancel
                        </button>
                      </div>
                    </div>
                  ) : (
                    <>
                      <div className="flex items-start justify-between gap-2">
                        <h3 className="min-w-0 flex-1 wrap-break-word font-semibold">
                          {image.name}
                        </h3>

                        <div className="flex shrink-0 items-center gap-1">
                          <button
                            type="button"
                            onClick={() => startEditing(image)}
                            disabled={saving || replacing || deletingId !== null}
                            title="Edit image"
                            aria-label={`Edit ${image.name}`}
                            className="rounded-lg p-2 text-neutral-500 transition hover:bg-neutral-100 hover:text-neutral-900 disabled:opacity-50"
                          >
                            <Pencil size={17} />
                          </button>

                          <button
                            type="button"
                            onClick={() => startReplacing(image)}
                            disabled={saving || replacing || deletingId !== null}
                            title="Replace image"
                            aria-label={`Replace ${image.name}`}
                            className="rounded-lg p-2 text-neutral-500 transition hover:bg-neutral-100 hover:text-neutral-900 disabled:opacity-50"
                          >
                            <RefreshCw size={17} />
                          </button>

                          <button
                            type="button"
                            onClick={() => void handleDelete(image)}
                            disabled={saving || replacing || deletingId !== null}
                            title="Delete image"
                            aria-label={`Delete ${image.name}`}
                            className="rounded-lg p-2 text-neutral-400 transition hover:bg-red-50 hover:text-red-600 disabled:opacity-50"
                          >
                            <Trash2 size={17} />
                          </button>
                        </div>
                      </div>

                      <div className="mt-3 flex flex-wrap gap-2">
                        {image.productType && (
                          <span className="rounded-full bg-neutral-100 px-3 py-1 text-xs text-neutral-600">
                            {image.productType}
                          </span>
                        )}

                        {image.color && (
                          <span className="rounded-full bg-neutral-100 px-3 py-1 text-xs text-neutral-600">
                            {image.color}
                          </span>
                        )}
                      </div>

                      <p className="mt-4 text-xs text-neutral-400">
                        {new Date(image.createdAt).toLocaleDateString('en-GB')}
                      </p>
                    </>
                  )}
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </main>
  )
}

import 'server-only'
import { prisma } from '@/lib/prisma'
import { isBrandId, type BrandId } from '@/lib/brands'
import { defaultProductMetadata, type ProductMetadata } from '@/lib/productMetadataTypes'
import type { Artwork } from '@/lib/artworks'

// Compatibility for older callers. Product fields are never stored on Artwork.
export async function listArtworks(): Promise<Artwork[]> {
  const records = await prisma.artwork.findMany({ orderBy: { createdAt: 'desc' } })
  return records.filter((item) => isBrandId(item.brand)).map((item) => ({
    id: item.id, brand: item.brand as BrandId, filename: item.filename,
    createdAt: item.createdAt.toISOString(), hasMockup: false,
    mockupColor: null, metadata: { ...defaultProductMetadata }
  }))
}

export async function getArtworkById(brand: BrandId, id: string): Promise<Artwork | null> {
  const item = await prisma.artwork.findFirst({ where: { id, brand } })
  return item ? {
    id: item.id, brand, filename: item.filename,
    createdAt: item.createdAt.toISOString(), hasMockup: false,
    mockupColor: null, metadata: { ...defaultProductMetadata }
  } : null
}

export async function getArtworkLinkedProducts(brand: BrandId, artworkId: string) {
  return prisma.product.findMany({
    where: { brand, artworkId },
    select: { id: true, title: true, slug: true, productType: true, status: true, createdAt: true },
    orderBy: { createdAt: 'desc' }
  })
}

import 'server-only'

import path from 'node:path'

export interface Artwork {
  id: string
  brand: BrandId
  filename: string
  createdAt: string
  hasMockup: boolean
  metadata: ProductMetadata
  mockupColor: string | null
}

const UUID_REGEX = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i

export function isArtworkId(id: string) {
  return UUID_REGEX.test(id)
}

export function getArtworkDirectory(
  brand: BrandId,
  id: string
) {
  if (!isArtworkId(id)) {
    throw new Error('Invalid artwork ID')
  }

  return path.join(
    process.cwd(),
    'designs',
    brand,
    id
  )
}


async function fileExists(filePath: string): Promise<boolean> {
  try {
    await stat(filePath)
    return true
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === 'ENOENT') {
      return false
    }

    throw error
  }
}

async function findArtworkMockup(directory: string, metadata: ProductMetadata): Promise<string | null> {
  const product = getProductType(metadata.productType)

  if (!product) return null

  for (const color of metadata.colors) {
    if (!product.colors.some((item) => item.id === color)) {
      continue
    }

    const filename = `${product.id}-${color}-front.webp`

    if (
      await fileExists(
        path.join(directory, 'mockups', filename)
      )
    ) {
      return color
    }
  }

  // Support older black shirt mockups.
  if (
    product.id === 'classic-tee' &&
    metadata.colors.includes('black') &&
    await fileExists(path.join(directory, 'shirt.webp'))
  ) {
    return 'black'
  }

  return null
}

export async function getArtworks(): Promise<Artwork[]> {
  const artworks: Artwork[] = []

  for (const brand of brands) {
    const brandDir = path.join(
      process.cwd(),
      'designs',
      brand.id
    )

    let entries

    try {
      entries = await readdir(brandDir, {
        withFileTypes: true,
      })
    } catch (error) {
      if (
        (error as NodeJS.ErrnoException).code === 'ENOENT'
      ) {
        continue
      }

      throw error
    }

    for (const entry of entries) {
      if (!entry.isDirectory() || !isArtworkId(entry.name)) {
        continue
      }

      const directory = getArtworkDirectory(
        brand.id,
        entry.name
      )

      try {
        const artworkStat = await stat(
          path.join(directory, 'artwork.png')
        )

        const metadata = await getProductMetadata(
          brand.id,
          entry.name
        )

        const mockupColor = await findArtworkMockup(
          directory,
          metadata
        )

        const hasMockup = mockupColor !== null

        artworks.push({
          id: entry.name,
          brand: brand.id,
          filename: 'artwork.png',
          createdAt: artworkStat.birthtime.toISOString(),
          hasMockup,
          mockupColor,
          metadata
        })
      } catch (error) {
        if (
          (error as NodeJS.ErrnoException).code !== 'ENOENT'
        ) {
          throw error
        }
      }
    }
  }

  return artworks.sort(
    (a, b) =>
      new Date(b.createdAt).getTime() -
      new Date(a.createdAt).getTime()
  )
}

import 'server-only'
import sharp from 'sharp'
import path from 'node:path'
import type { BrandId } from '@/lib/brands'
import { getProductType } from '@/lib/catalog/products'
import {
  getArtworkImage,
  uploadArtworkMockup
} from '@/lib/artworkStorage'

const OUTPUT = {
  width: 1000,
  height: 1200,
  quality: 85
}

export async function generateMockup(
  brand: BrandId,
  id: string,
  productTypeId = 'classic-tee',
  colorId = 'black'
) {
  const productType = getProductType(productTypeId)

  if (!productType) {
    throw new Error(`Unknown product type: ${productTypeId}`)
  }

  const color = productType.colors.find(
    (item) => item.id === colorId
  )

  if (!color) {
    throw new Error(
      `Color "${colorId}" is not available for ${productTypeId}`
    )
  }

  const shirtPath = path.join(
    process.cwd(),
    'assets',
    'mockups',
    color.mockupTemplate
  )

  const filename = `${productTypeId}-${colorId}-front.webp`
  const printArea = productType.printArea

  // Download original artwork from Supabase Storage.
  const originalArtwork = await getArtworkImage(brand, id)

  // Prepare the garment template.
  const shirt = await sharp(shirtPath)
    .resize(OUTPUT.width, OUTPUT.height, {
      fit: 'contain',
      background: {
        r: 0,
        g: 0,
        b: 0,
        alpha: 0
      }
    })
    .ensureAlpha()
    .png()
    .toBuffer()

  // Fit artwork inside the print area.
  const artwork = await sharp(originalArtwork)
    .trim({
      background: '#00000000',
      threshold: 5
    })
    .resize(printArea.width, printArea.height, {
      fit: 'contain',
      background: {
        r: 0,
        g: 0,
        b: 0,
        alpha: 0
      }
    })
    .png()
    .toBuffer()

  // Combine artwork and garment.
  const mockupBuffer = await sharp(shirt)
    .composite([
      {
        input: artwork,
        left: printArea.left,
        top: printArea.top
      }
    ])
    .webp({ quality: OUTPUT.quality })
    .toBuffer()

  // Upload generated mockup to Supabase Storage.
  const storagePath = await uploadArtworkMockup(
    brand,
    id,
    filename,
    mockupBuffer
  )

  return {
    storagePath,
    filename,
    productType: productTypeId,
    color: colorId
  }
}

import 'server-only'
import path from 'node:path'
import { readFile, writeFile } from 'node:fs/promises'
import type { BrandId } from '@/lib/brands'
import { getArtworkDirectory } from '@/lib/artworks'
import { productOptions } from '@/lib/productOptions'
import { defaultProductMetadata, type ProductMetadata } from '@/lib/productMetadataTypes'
import { getProductType } from '@/lib/catalog/products'
import { prisma } from '@/lib/prisma'

export async function getProductMetadata( brand: BrandId, id: string ): Promise<ProductMetadata> {
  const filePath = path.join(
    getArtworkDirectory(brand, id),
    'metadata.json'
  )

  try {
    const content = await readFile(filePath, 'utf8')
    const saved = JSON.parse(content) as Partial<ProductMetadata>

    return {
      ...defaultProductMetadata,
      ...saved,
      colors: Array.isArray(saved.colors) ? saved.colors : ['black'],
      sizes: Array.isArray(saved.sizes) ? saved.sizes : [],
      collections: Array.isArray(saved.collections) ? saved.collections : []
    }
  } catch (error) {
    if (
      (error as NodeJS.ErrnoException).code === 'ENOENT'
    ) {
      return {
        ...defaultProductMetadata,
        sizes: [],
        collections: []
      }
    }

    throw error
  }
}


export function validateProductMetadata( brand: BrandId, value: unknown ): value is ProductMetadata {
  if (!value || typeof value !== 'object') return false

  const data = value as Record<string, unknown>
  const options = productOptions[brand]

  const productType =
  typeof data.productType === 'string'
    ? getProductType(data.productType)
    : undefined

  return (
    typeof data.title === 'string' &&
    data.title.length <= 150 &&

    typeof data.slug === 'string' &&
    (data.slug === '' ||
      /^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(data.slug)) &&
    data.slug.length <= 180 &&

    typeof data.description === 'string' &&
    data.description.length <= 5000 &&

    typeof data.price === 'number' &&
      Number.isFinite(data.price) &&
      data.price >= 0 &&
      Math.abs(data.price * 100 - Math.round(data.price * 100)) < 0.000001 &&

    Array.isArray(data.sizes) &&
      data.sizes.every(
        (size) =>
          typeof size === 'string' &&
          productType?.sizes.includes(size)
      ) &&

    typeof data.category === 'string' &&
    (data.category === '' ||
      options.categories.some(
        (item) => item.id === data.category
      )) &&

    Array.isArray(data.collections) &&
    data.collections.every(
      (id) =>
        typeof id === 'string' &&
        options.collections.some((item) => item.id === id)
    ) &&

    Boolean(productType) &&

    Array.isArray(data.colors) &&
    data.colors.length > 0 &&
    data.colors.every(
      (color) =>
        typeof color === 'string' &&
        productType?.colors.some(
          (available) => available.id === color
        )
    ) &&

    (data.status === 'draft' || data.status === 'ready')
  )
}

export async function saveProductMetadata(brand: BrandId, id: string, metadata: ProductMetadata) {
  const filePath = path.join(
    getArtworkDirectory(brand, id),
    'metadata.json'
  )

  await writeFile(
    filePath,
    JSON.stringify(metadata, null, 2),
    'utf8'
  )
}

export async function isSlugTaken(
  brand: BrandId,
  slug: string,
  excludeArtworkId?: string
): Promise<boolean> {
  if (!slug) return false

  const existingProduct = await prisma.product.findFirst({
    where: {
      brand,
      slug,
      ...(excludeArtworkId
        ? {
            NOT: {
              artworkId: excludeArtworkId
            }
          }
        : {})
    },
    select: {
      id: true
    }
  })

  return existingProduct !== null
}

export function slugify(value: string): string {
  return value
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
}

import { createServerClient } from '@supabase/ssr'
import { NextResponse, type NextRequest } from 'next/server'

export async function updateSession(request: NextRequest) {
  let response = NextResponse.next({ request })

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll()
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) => {
            request.cookies.set(name, value)
          })

          response = NextResponse.next({ request })

          cookiesToSet.forEach(({ name, value, options }) => {
            response.cookies.set(name, value, options)
          })
        }
      }
    }
  )

  // Validate and refresh the authentication session.
  await supabase.auth.getUser()

  return response
}

import 'dotenv/config'
import { PrismaClient } from '@/lib/generated/prisma/client'

const prisma = new PrismaClient()

async function main() {
  console.log('Testing database connection...')

  // Test database connection
  await prisma.$connect()
  console.log('Database connected!')

  // Create a temporary artwork
  const artwork = await prisma.artwork.create({
    data: {
      brand: 't4c',
      filename: 'artwork.png',
    },
  })

  console.log('Artwork created:', artwork.id)

  try {
    // Retrieve the artwork
    const saved = await prisma.artwork.findUnique({
      where: { id: artwork.id },
    })

    if (!saved) {
      throw new Error('Artwork could not be retrieved.')
    }

    console.log('Artwork retrieved:', saved.filename)
  } finally {
    // Always remove the temporary artwork
    await prisma.artwork.delete({
      where: { id: artwork.id },
    })

    console.log('Test artwork removed.')
  }

  console.log('Test completed! Database connection works.')
}

main()
  .catch((error) => {
    console.error('Database test failed:', error)
    process.exitCode = 1
  })
  .finally(async () => {
    await prisma.$disconnect()
  })

========== scripts/upload-artwork.ts ==========

import 'dotenv/config'
import path from 'node:path'
import { readFile, readdir } from 'node:fs/promises'
import { createClient } from '@supabase/supabase-js'

const brand = 'legends'
const artworkId = '780a50fa-7c6e-4345-bbe1-27885576df94'

async function main() {
  const url = process.env.SUPABASE_URL
  const secretKey = process.env.SUPABASE_SECRET_KEY

  if (!url || !secretKey) {
    throw new Error('Missing Supabase environment variables')
  }

  const supabase = createClient(url, secretKey, {
    auth: {
      autoRefreshToken: false,
      persistSession: false
    }
  })

  const directory = path.join(
    process.cwd(),
    'designs',
    brand,
    artworkId
  )

  async function uploadFile(
    localPath: string,
    storagePath: string,
    contentType: string
  ) {
    const file = await readFile(localPath)

    const { error } = await supabase.storage
      .from('artworks')
      .upload(storagePath, file, {
        contentType,
        upsert: false,
      })

    if (error) {
      // Already uploaded? Leave the existing file untouched.
      if (error.statusCode === '409') {
        console.log('Already exists, skipping:', storagePath)
        return
      }

      throw error
    }

    console.log('Uploaded:', storagePath)
  }

  // Original artwork
  await uploadFile(
    path.join(directory, 'artwork.png'),
    `${brand}/${artworkId}/artwork.png`,
    'image/png'
  )

  // Mockups
  const mockupDirectory = path.join(directory, 'mockups')

  const files = await readdir(mockupDirectory)

  for (const filename of files) {
    if (!filename.endsWith('.webp')) continue

    await uploadFile(
      path.join(mockupDirectory, filename),
      `${brand}/${artworkId}/mockups/${filename}`,
      'image/webp'
    )
  }

  console.log('Artwork and mockups upload completed!')
}

main().catch((error) => {
  console.error('Upload failed:', error)
  process.exitCode = 1
})

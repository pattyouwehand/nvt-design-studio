import 'server-only'
import { prisma } from '@/lib/prisma'
import { isBrandId, type BrandId } from '@/lib/brands'

  defaultProductMetadata,
  type ProductMetadata
} from '@/lib/productMetadataTypes'

function getStoredMetadata(value: Prisma.JsonValue): ProductMetadata {
  return {
    ...defaultProductMetadata,
    ...(value as Partial<ProductMetadata>)
  }
}

function getProductMetadata(
  stored: ProductMetadata,
  product: {
    title: string
    slug: string
    subtitle: string | null
    description: string | null
    price: number | null
    sizes: Prisma.JsonValue
    colors: Prisma.JsonValue
    category: string
    label: string
    collections: Prisma.JsonValue
    status: string
    productType: string
  } | null
): ProductMetadata {
  if (!product) return stored

  return {
    ...stored,
    title: product.title,
    slug: product.slug,
    subtitle: product.subtitle ?? '',
    description: product.description ?? '',
    price: product.price ?? 0,
    sizes: Array.isArray(product.sizes)
      ? product.sizes as string[]
      : [],
    colors: Array.isArray(product.colors)
      ? product.colors as string[]
      : [],
    category: product.category,
    label: product.label,
    collections: Array.isArray(product.collections)
      ? product.collections as string[]
      : [],
    status: product.status === 'ready' ? 'ready' : 'draft',
    productType: product.productType
  }
}

async function findMockupColor(
  brand: BrandId,
  id: string,
  metadata: ProductMetadata
): Promise<string | null> {
  const product = getProductType(metadata.productType)

  if (!product) return null

  const folder = `${brand}/${id}/mockups`

  const { data: files, error } = await supabaseAdmin.storage
    .from('artworks')
    .list(folder, { limit: 100 })

  if (error) throw error

  for (const color of metadata.colors) {
    if (!product.colors.some((item) => item.id === color)) {
      continue
    }

    const filename = `${product.id}-${color}-front.webp`

    if (files.some((file) => file.name === filename)) {
      return color
    }
  }

  return null
}

export async function listArtworks(): Promise<Artwork[]> {
  const records = await prisma.artwork.findMany({
    include: {
      products: {
        orderBy: { createdAt: 'asc' },
        take: 1
      }
    },
    orderBy: { createdAt: 'desc' }
  })

  return Promise.all(
    records
      .filter((record) => isBrandId(record.brand))
      .map(async (record): Promise<Artwork> => {
        const stored = getStoredMetadata(record.metadata)
        const metadata = getProductMetadata(
          stored,
          record.products[0] ?? null
        )

        const mockupColor = await findMockupColor(
          record.brand as BrandId,
          record.id,
          metadata
        )

        return {
          id: record.id,
          brand: record.brand as BrandId,
          filename: record.filename,
          createdAt: record.createdAt.toISOString(),
          hasMockup: mockupColor !== null,
          metadata,
          mockupColor
        }
      })
  )
}

export async function getArtworkById(
  brand: BrandId,
  id: string
): Promise<Artwork | null> {
  const record = await prisma.artwork.findUnique({
    where: { id },
    include: {
      products: {
        orderBy: { createdAt: 'asc' },
        take: 1
      }
    }
  })

  if (!record || record.brand !== brand) {
    return null
  }

  const metadata = getProductMetadata(
    getStoredMetadata(record.metadata),
    record.products[0] ?? null
  )

  const mockupColor = await findMockupColor(
    brand,
    id,
    metadata
  )

  return {
    id: record.id,
    brand,
    filename: record.filename,
    createdAt: record.createdAt.toISOString(),
    hasMockup: mockupColor !== null,
    metadata,
    mockupColor
  }
}

export async function updateArtworkMetadata(
  brand: BrandId,
  id: string,
  metadata: ProductMetadata,
  productId?: string
): Promise<boolean> {
  const artwork = await prisma.artwork.findFirst({
    where: { id, brand },
    select: { id: true }
  })

  if (!artwork) return false

  const existingProduct = productId
    ? await prisma.product.findFirst({
        where: {
          id: productId,
          artworkId: id,
          brand
        }
      })
    : await prisma.product.findFirst({
        where: {
          artworkId: id,
          brand
        },
        orderBy: { createdAt: 'asc' }
      })

  // Never create a replacement when an explicit product ID is invalid.
  if (productId && !existingProduct) {
    throw new Error('Product not found for this artwork.')
  }

  await prisma.$transaction(async (tx) => {
    const productData = {
      brand,
      productType: metadata.productType,
      title: metadata.title,
      slug: metadata.slug,
      subtitle: metadata.subtitle,
      description: metadata.description,
      price: metadata.price,
      status: metadata.status,
      sizes: metadata.sizes,
      colors: metadata.colors,
      category: metadata.category,
      label: metadata.label,
      collections: metadata.collections,
      metadata: metadata as unknown as Prisma.InputJsonValue
    }

    if (existingProduct) {
      await tx.product.update({
        where: { id: existingProduct.id },
        data: productData
      })
    } else if (metadata.title.trim() && metadata.slug.trim()) {
      await tx.product.create({
        data: {
          ...productData,
          artworkId: id
        }
      })
    }

    // Keep legacy artwork metadata only for the original product.
    const firstProduct = await tx.product.findFirst({
      where: { artworkId: id, brand },
      orderBy: { createdAt: 'asc' },
      select: { id: true }
    })

    if (!firstProduct || firstProduct.id === existingProduct?.id || !productId) {
      await tx.artwork.update({
        where: { id },
        data: {
          metadata: metadata as unknown as Prisma.InputJsonValue
        }
      })
    }
  })

  return true
}

export async function isArtworkSlugTaken(
  brand: BrandId,
  slug: string,
  excludeId: string
): Promise<boolean> {
  const existing = await prisma.product.findFirst({
    where: {
      brand,
      slug,
      artworkId: { not: excludeId }
    },
    select: { id: true }
  })

  if (existing) return true

  // Also check legacy artworks without migrated products.
  const records = await prisma.artwork.findMany({
    where: {
      brand,
      id: { not: excludeId },
      products: { none: {} }
    },
    select: { metadata: true }
  })

  return records.some((record) => {
    const metadata = record.metadata as Partial<ProductMetadata>
    return metadata?.slug === slug
  })
}

export async function deleteArtwork(
  brand: BrandId,
  id: string
): Promise<boolean> {
  const linkedProducts = await prisma.product.count({
    where: { artworkId: id, brand }
  })

  if (linkedProducts > 0) {
    return false
  }

  const result = await prisma.artwork.deleteMany({
    where: { id, brand }
  })

  return result.count > 0
}

const UUID_REGEX = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i

export function isArtworkId(id: string) {
  return UUID_REGEX.test(id)
}

export function getArtworkDirectory(
  brand: BrandId,
  id: string
) {
  if (!isArtworkId(id)) {
    throw new Error('Invalid artwork ID')
  }

  return path.join(
    process.cwd(),
    'designs',
    brand,
    id
  )
}


async function fileExists(filePath: string): Promise<boolean> {
  try {
    await stat(filePath)
    return true
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === 'ENOENT') {
      return false
    }

    throw error
  }
}

async function findArtworkMockup(directory: string, metadata: ProductMetadata): Promise<string | null> {
  const product = getProductType(metadata.productType)

  if (!product) return null

  for (const color of metadata.colors) {
    if (!product.colors.some((item) => item.id === color)) {
      continue
    }

    const filename = `${product.id}-${color}-front.webp`

    if (
      await fileExists(
        path.join(directory, 'mockups', filename)
      )
    ) {
      return color
    }
  }

  // Support older black shirt mockups.
  if (
    product.id === 'classic-tee' &&
    metadata.colors.includes('black') &&
    await fileExists(path.join(directory, 'shirt.webp'))
  ) {
    return 'black'
  }

  return null
}

export async function getArtworks(): Promise<Artwork[]> {
  const artworks: Artwork[] = []

  for (const brand of brands) {
    const brandDir = path.join(
      process.cwd(),
      'designs',
      brand.id
    )

    let entries

    try {
      entries = await readdir(brandDir, {
        withFileTypes: true,
      })
    } catch (error) {
      if (
        (error as NodeJS.ErrnoException).code === 'ENOENT'
      ) {
        continue
      }

      throw error
    }

    for (const entry of entries) {
      if (!entry.isDirectory() || !isArtworkId(entry.name)) {
        continue
      }

      const directory = getArtworkDirectory(
        brand.id,
        entry.name
      )

      try {
        const artworkStat = await stat(
          path.join(directory, 'artwork.png')
        )

        const metadata = await getProductMetadata(
          brand.id,
          entry.name
        )

        const mockupColor = await findArtworkMockup(
          directory,
          metadata
        )

        const hasMockup = mockupColor !== null

        artworks.push({
          id: entry.name,
          brand: brand.id,
          filename: 'artwork.png',
          createdAt: artworkStat.birthtime.toISOString(),
          hasMockup,
          mockupColor,
          metadata
        })
      } catch (error) {
        if (
          (error as NodeJS.ErrnoException).code !== 'ENOENT'
        ) {
          throw error
        }
      }
    }
  }

  return artworks.sort(
    (a, b) =>
      new Date(b.createdAt).getTime() -
      new Date(a.createdAt).getTime()
  )
}

import 'server-only'
import path from 'node:path'

import type { BrandId } from '@/lib/brands'

import { getProductType } from '@/lib/catalog/products'
import { prisma } from '@/lib/prisma'

export async function getProductMetadata( brand: BrandId, id: string ): Promise<ProductMetadata> {
  const filePath = path.join(
    getArtworkDirectory(brand, id),
    'metadata.json'
  )

  try {
    const content = await readFile(filePath, 'utf8')
    const saved = JSON.parse(content) as Partial<ProductMetadata>

    return {
      ...defaultProductMetadata,
      ...saved,
      colors: Array.isArray(saved.colors) ? saved.colors : ['black'],
      sizes: Array.isArray(saved.sizes) ? saved.sizes : [],
      collections: Array.isArray(saved.collections) ? saved.collections : []
    }
  } catch (error) {
    if (
      (error as NodeJS.ErrnoException).code === 'ENOENT'
    ) {
      return {
        ...defaultProductMetadata,
        sizes: [],
        collections: []
      }
    }

    throw error
  }
}


export function validateProductMetadata( brand: BrandId, value: unknown ): value is ProductMetadata {
  if (!value || typeof value !== 'object') return false

  const data = value as Record<string, unknown>
  const options = productOptions[brand]

  const productType =
  typeof data.productType === 'string'
    ? getProductType(data.productType)
    : undefined

  return (
    typeof data.title === 'string' &&
    data.title.length <= 150 &&

    typeof data.slug === 'string' &&
    (data.slug === '' ||
      /^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(data.slug)) &&
    data.slug.length <= 180 &&

    typeof data.description === 'string' &&
    data.description.length <= 5000 &&

    typeof data.price === 'number' &&
      Number.isFinite(data.price) &&
      data.price >= 0 &&
      Math.abs(data.price * 100 - Math.round(data.price * 100)) < 0.000001 &&

    Array.isArray(data.sizes) &&
      data.sizes.every(
        (size) =>
          typeof size === 'string' &&
          productType?.sizes.includes(size)
      ) &&

    typeof data.category === 'string' &&
    (data.category === '' ||
      options.categories.some(
        (item) => item.id === data.category
      )) &&

    Array.isArray(data.collections) &&
    data.collections.every(
      (id) =>
        typeof id === 'string' &&
        options.collections.some((item) => item.id === id)
    ) &&

    Boolean(productType) &&

    Array.isArray(data.colors) &&
    data.colors.length > 0 &&
    data.colors.every(
      (color) =>
        typeof color === 'string' &&
        productType?.colors.some(
          (available) => available.id === color
        )
    ) &&

    (data.status === 'draft' || data.status === 'ready')
  )
}

export async function saveProductMetadata(brand: BrandId, id: string, metadata: ProductMetadata) {
  const filePath = path.join(
    getArtworkDirectory(brand, id),
    'metadata.json'
  )

  await writeFile(
    filePath,
    JSON.stringify(metadata, null, 2),
    'utf8'
  )
}

export async function isSlugTaken(
  brand: BrandId,
  slug: string,
  excludeArtworkId?: string
): Promise<boolean> {
  if (!slug) return false

  const existingProduct = await prisma.product.findFirst({
    where: {
      brand,
      slug,
      ...(excludeArtworkId
        ? {
            NOT: {
              artworkId: excludeArtworkId
            }
          }
        : {})
    },
    select: {
      id: true
    }
  })

  return existingProduct !== null
}

import { randomUUID } from 'node:crypto'
import { isBrandId } from '@/lib/brands'
import { listArtworks } from '@/lib/artworkDatabase'
import { uploadArtworkImage } from '@/lib/artworkStorage'
import { prisma } from '@/lib/prisma'
import { supabaseAdmin } from '@/lib/supabaseAdmin'
import { defaultProductMetadata } from '@/lib/productMetadataTypes'
import type { Prisma } from '@/lib/generated/prisma/client'
import { getStudioUser } from '@/lib/auth'

export const runtime = 'nodejs'

const MAX_FILE_SIZE = 25 * 1024 * 1024

export async function POST(request: NextRequest) {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const formData = await request.formData()

    const file = formData.get('artwork')
    const brand = formData.get('brand')

    if (typeof brand !== 'string' || !isBrandId(brand)) {
      return NextResponse.json(
        { error: 'Please select a valid brand.' },
        { status: 400 }
      )
    }

    if (!(file instanceof File)) {
      return NextResponse.json(
        { error: 'Please upload an artwork.' },
        { status: 400 }
      )
    }

    if (file.type !== 'image/png') {
      return NextResponse.json(
        { error: 'Only PNG files are supported.' },
        { status: 400 }
      )
    }

    if (file.size === 0 || file.size > MAX_FILE_SIZE) {
      return NextResponse.json(
        { error: 'File must be between 1 byte and 25 MB.' },
        { status: 400 }
      )
    }

    const buffer = Buffer.from(await file.arrayBuffer())
    const imageMetadata = await sharp(buffer).metadata()

    if (
      imageMetadata.format !== 'png' ||
      !imageMetadata.width ||
      !imageMetadata.height
    ) {
      return NextResponse.json(
        { error: 'Invalid PNG image.' },
        { status: 400 }
      )
    }

    if (
      imageMetadata.width > 10000 ||
      imageMetadata.height > 10000
    ) {
      return NextResponse.json(
        { error: 'Image dimensions are too large.' },
        { status: 400 }
      )
    }

    if (!imageMetadata.hasAlpha) {
      return NextResponse.json(
        { error: 'Artwork must have a transparency channel.' },
        { status: 400 }
      )
    }

    const id = randomUUID()
    const storagePath = await uploadArtworkImage(
      brand,
      id,
      buffer
    )

    try {
      await prisma.artwork.create({
        data: {
          id,
          brand,
          filename: file.name,
          metadata: defaultProductMetadata as unknown as Prisma.InputJsonValue
        }
      })
    } catch (databaseError) {
      // Remove the uploaded PNG if the database insert fails.
      const { error: cleanupError } = await supabaseAdmin.storage
        .from('artworks')
        .remove([storagePath])

      if (cleanupError) {
        console.error(
          'Unable to clean up uploaded artwork:',
          cleanupError
        )
      }

      throw databaseError
    }

    return NextResponse.json(
      {
        success: true,
        id,
        brand,
        filename: file.name,
        width: imageMetadata.width,
        height: imageMetadata.height
      },
      { status: 201 }
    )
  } catch (error) {
    console.error('Artwork upload failed:', error)

    return NextResponse.json(
      { error: 'Unable to upload artwork.' },
      { status: 500 }
    )
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'
import { getArtworkById, updateArtworkMetadata, isArtworkSlugTaken, deleteArtwork } from '@/lib/artworkDatabase'
import { validateProductMetadata } from '@/lib/productMetadata'
import { getStudioUser } from '@/lib/auth'
import { deleteArtworkFiles } from '@/lib/artworkStorage'
import { prisma } from '@/lib/prisma'
import { defaultProductMetadata } from '@/lib/productMetadataTypes'

export const runtime = 'nodejs'

export async function GET(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params
  const brand = request.nextUrl.searchParams.get('brand')
  const selectedProductId = request.nextUrl.searchParams.get('productId')
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  if (!brand || !isBrandId(brand) || !isArtworkId(id)) {
    return NextResponse.json(
      { error: 'Invalid artwork.' },
      { status: 400 }
    )
  }

  try {
    const artwork = await getArtworkById(brand, id)

    if (!artwork) {
      return NextResponse.json(
        { error: 'Artwork not found.' },
        { status: 404 }
      )
    }

    // Fetch every product linked to this artwork.
    const linkedProducts = await prisma.product.findMany({
      where: {
        artworkId: id,
        brand
      },
      select: {
        id: true,
        title: true,
        slug: true,
        productType: true,
        status: true,
        createdAt: true
      },
      orderBy: {
        createdAt: 'asc'
      }
    })

    // Fetch published product IDs in one query.
    const publishedProducts = linkedProducts.length
      ? await prisma.publishedProduct.findMany({
          where: {
            productId: {
              in: linkedProducts.map((product) => product.id)
            }
          },
          select: {
            productId: true
          }
        })
      : []

    const publishedIds = new Set(
      publishedProducts.map((product) => product.productId)
    )

    const products = linkedProducts.map((product) => ({
      ...product,
      isPublished: publishedIds.has(product.id)
    }))

    const selectedProduct = selectedProductId ? products.find((product) => product.id === selectedProductId) : products[0]

    if (selectedProductId && !selectedProduct) {
      return NextResponse.json(
        { error: 'Product not found for this artwork.' },
        { status: 404 }
      )
    }

    const selectedProductRecord = selectedProduct
      ? await prisma.product.findUnique({
          where: { id: selectedProduct.id }
        })
      : null

    const selectedMetadata = selectedProductRecord
      ? {
          ...defaultProductMetadata,
          title: selectedProductRecord.title,
          slug: selectedProductRecord.slug,
          subtitle: selectedProductRecord.subtitle ?? '',
          description: selectedProductRecord.description ?? '',
          price: selectedProductRecord.price ?? 0,
          sizes: selectedProductRecord.sizes,
          colors: selectedProductRecord.colors,
          category: selectedProductRecord.category,
          label: selectedProductRecord.label,
          collections: selectedProductRecord.collections,
          status: selectedProductRecord.status,
          productType: selectedProductRecord.productType
        }
      : artwork.metadata

    return NextResponse.json({
      ...artwork,
      metadata: selectedMetadata,
      productId: selectedProduct?.id ?? null,
      isPublished: selectedProduct?.isPublished ?? false,
      linkedProducts: products,
      width: null,
      height: null
    })
  } catch (error) {
    console.error('Unable to load artwork:', error)

    return NextResponse.json(
      { error: 'Unable to load artwork.' },
      { status: 500 }
    )
  }
}

export async function PUT(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params
  const brand = request.nextUrl.searchParams.get('brand')
  const productId = request.nextUrl.searchParams.get('productId')
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  if (!brand || !isBrandId(brand) || !isArtworkId(id)) {
    return NextResponse.json(
      { error: 'Invalid artwork.' },
      { status: 400 }
    )
  }

  try {
    const metadata: unknown = await request.json()

    if (!validateProductMetadata(brand, metadata)) {
      return NextResponse.json(
        { error: 'Invalid product details.' },
        { status: 400 }
      )
    }

    // When editing an existing product, require its ID.
    const existingProducts = await prisma.product.findMany({
      where: { artworkId: id, brand },
      select: { id: true }
    })

    if (existingProducts.length > 0 && !productId) {
      return NextResponse.json(
        { error: 'Select a product before saving.' },
        { status: 400 }
      )
    }

    if (
      productId &&
      !existingProducts.some((product) => product.id === productId)
    ) {
      return NextResponse.json(
        { error: 'Product not found for this artwork.' },
        { status: 404 }
      )
    }

    // Check slug conflicts, excluding the selected product.
    if (metadata.slug) {
      const slugConflict = await prisma.product.findFirst({
        where: {
          brand,
          slug: metadata.slug,
          ...(productId ? { id: { not: productId } } : {})
        },
        select: { id: true }
      })

      if (slugConflict) {
        return NextResponse.json(
          { error: 'This URL slug is already in use for this brand.' },
          { status: 409 }
        )
      }
    }

    const updated = await updateArtworkMetadata(
      brand,
      id,
      metadata,
      productId ?? undefined
    )

    if (!updated) {
      return NextResponse.json(
        { error: 'Artwork not found.' },
        { status: 404 }
      )
    }

    return NextResponse.json({
      success: true,
      metadata
    })
  } catch (error) {
    console.error('Unable to save product details:', error)

    return NextResponse.json(
      { error: 'Unable to save product details.' },
      { status: 500 }
    )
  }
}

  _request: Request,
  { params }: { params: Promise<{ id: string }> }

    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const { id } = await params

  if (!isArtworkId(id)) {
    return NextResponse.json(
      { error: 'Invalid product ID.' },
      { status: 400 }
    )
  }

  try {
    const product = await prisma.product.findUnique({
      where: { id },
      include: { artwork: true }
    })

    if (!product) {
      return NextResponse.json(

    }

      // Remove the storefront publication, if present.

        where: { productId: id }
      })

      // ProductSharedImage links are deleted automatically
      // through the Prisma cascade relation.
      await tx.product.delete({
        where: { id }
      })

      // Reset legacy metadata only when this artwork has
      // no other products.
      const remainingProducts = await tx.product.count({
        where: { artworkId: product.artworkId }
      })

      if (remainingProducts === 0) {
        await tx.artwork.update({
          where: { id: product.artworkId },
          data: {
            metadata: defaultProductMetadata as unknown as Prisma.InputJsonValue
          }
        })
      }
    })

    return NextResponse.json({
      success: true,
      artworkId: product.artworkId
    })
  } catch (error) {
    console.error('Product deletion failed:', error)

    return NextResponse.json(
      { error: 'Unable to delete product.' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getStudioUser } from '@/lib/auth'
import { getProductType } from '@/lib/catalog/products'
import { getArtworkMockup } from '@/lib/artworkStorage'
import { supabaseAdmin } from '@/lib/supabaseAdmin'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'

export const runtime = 'nodejs'

const PUBLIC_BUCKET = 'product-images'

export async function POST(_request: Request, { params }: { params: Promise<{ id: string }> }) {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const { id } = await params

  if (!isArtworkId(id)) {
    return NextResponse.json(
      { error: 'Invalid product ID.' },
      { status: 400 }
    )
  }

  try {
    const product = await prisma.product.findUnique({
      where: { id },
      include: { artwork: true }
    })

    if (!product) {
      return NextResponse.json(
        { error: 'Product not found.' },
        { status: 404 }
      )
    }

    const brand = product.brand

    if (
      !isBrandId(brand) ||
      product.artwork.brand !== brand
    ) {
      return NextResponse.json(
        { error: 'Invalid product brand.' },
        { status: 400 }
      )
    }

    const productType = getProductType(product.productType)

    if (!productType) {
      return NextResponse.json(
        { error: 'Unknown product type.' },
        { status: 400 }
      )
    }

    const sizes = product.sizes
    const colors = product.colors
    const collections = product.collections

    if (
      product.status !== 'ready' ||
      !product.title.trim() ||
      !product.slug.trim() ||
      product.price === null ||
      !Number.isFinite(product.price) ||
      product.price <= 0 ||
      !Array.isArray(sizes) ||
      sizes.length === 0 ||
      !sizes.every((size) => typeof size === 'string') ||
      !Array.isArray(colors) ||
      colors.length === 0 ||
      !colors.every((color) => typeof color === 'string') ||
      !colors.every((color) =>
        productType.colors.some((item) => item.id === color)
      )
    ) {
      return NextResponse.json(
        { error: 'Complete the product details before publishing.' },
        { status: 400 }
      )
    }

    // Prevent another product from using the same brand/slug.
    const slugConflict = await prisma.publishedProduct.findFirst({
      where: {
        brand,
        slug: product.slug,
        productId: { not: product.id }
      },
      select: { id: true }
    })

    if (slugConflict) {
      return NextResponse.json(
        { error: 'This URL slug is already published for this brand.' },
        { status: 409 }
      )
    }

    const images: Record<string, string> = {}
    const uploadedPaths: string[] = []

    // Upload public mockups before updating the published snapshot.
    for (const color of colors as string[]) {
      const filename = `${productType.id}-${color}-front.webp`

      // Read from the existing private artworks bucket.
      const mockup = await getArtworkMockup(
        brand,
        product.artworkId,
        filename
      )

      // Product ID ensures different clothing types do not collide.
      const storagePath = `${brand}/${product.id}/${filename}`

      const { error: uploadError } = await supabaseAdmin.storage
        .from(PUBLIC_BUCKET)
        .upload(storagePath, mockup, {
          contentType: 'image/webp',
          upsert: true,
          cacheControl: '60'
        })

      if (uploadError) {
        throw uploadError
      }

      uploadedPaths.push(storagePath)

      const { data } = supabaseAdmin.storage
        .from(PUBLIC_BUCKET)
        .getPublicUrl(storagePath)

      // Version query helps browsers refresh updated mockups.
      images[color] = `${data.publicUrl}?v=${Date.now()}`
    }

    const snapshot = {
      id: product.id,
      artworkId: product.artworkId,
      brand,
      productType: product.productType,
      title: product.title,
      slug: product.slug,
      subtitle: product.subtitle,
      description: product.description ?? '',
      price: product.price,
      sizes,
      colors,
      category: product.category,
      collections: Array.isArray(collections) ? collections : [],
      images
    }

    const published = await prisma.publishedProduct.upsert({
      where: { productId: product.id },
      create: {
        productId: product.id,
        brand,
        slug: product.slug,
        data: snapshot
      },
      update: {
        brand,
        slug: product.slug,
        data: snapshot
      }
    })

    return NextResponse.json({
      success: true,
      productId: published.productId,
      publishedAt: published.publishedAt,
      images: uploadedPaths.length
    })
  } catch (error) {
    console.error('Product publishing failed:', error)

    return NextResponse.json(
      {
        error: 'Unable to publish product. Check that all selected colors have generated mockups.'
      },
      { status: 500 }
    )
  }
}

export async function GET(request: NextRequest) {
  const { searchParams } = request.nextUrl
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const brand = searchParams.get('brand')
  const slug = searchParams.get('slug')
  const artworkId = searchParams.get('artworkId')

  if (
    !brand ||
    !isBrandId(brand) ||
    !slug ||
    !/^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(slug) ||
    (artworkId !== null && !isArtworkId(artworkId))
  ) {
    return NextResponse.json(
      { error: 'Invalid slug check parameters.' },
      { status: 400 }
    )
  }

  try {
    const taken = await isSlugTaken(
      brand,
      slug,
      artworkId ?? undefined
    )

    return NextResponse.json({ available: !taken })
  } catch (error) {
    console.error('Unable to check slug:', error)

    return NextResponse.json(
      { error: 'Unable to check slug.' },
      { status: 500 }
    )
  }
}

  createdAt: string
  hasMockup: boolean
  metadata: ProductMetadata
  productId: string | null
  isPublished: boolean
  linkedProducts: LinkedProduct[]
}

export default function ArtworkDetailPage({ params }: { params: Promise<{ id: string }> }) {
  const { id } = use(params)
  const router = useRouter()

  const searchParams = useSearchParams()
  const brand = searchParams.get('brand')
  const selectedProductId = searchParams.get('productId')

  const [artwork, setArtwork] = useState<ArtworkDetail | null>(null)
  const [loading, setLoading] = useState(true)
  const [generating, setGenerating] = useState(false)
  const [error, setError] = useState('')
  const [message, setMessage] = useState('')
  const [mockupVersion, setMockupVersion] = useState(0)
  const [selectedColor, setSelectedColor] = useState('black')
  const [availableMockups, setAvailableMockups] = useState<string[]>([])
  const [checkingMockups, setCheckingMockups] = useState(true)
  const [details, setDetails] = useState<ProductMetadata>(defaultProductMetadata)
  const [saving, setSaving] = useState(false)
  const [savedDetails, setSavedDetails] = useState<ProductMetadata | null>(null)
  const [exporting, setExporting] = useState(false)
  const [priceInput, setPriceInput] = useState('')
  const [slugManuallyEdited, setSlugManuallyEdited] = useState(false)
  const [slugAvailable, setSlugAvailable] = useState<boolean | null>(null)
  const [checkingSlug, setCheckingSlug] = useState(false)
  const [deleting, setDeleting] = useState(false)
  const [deletingProduct, setDeletingProduct] = useState(false)
  const [publishing, setPublishing] = useState(false)
  const [unpublishing, setUnpublishing] = useState(false)
  const [creatingProduct, setCreatingProduct] = useState(false)

  useEffect(() => {
  async function loadArtwork() {
    if (!brand) {
      setError('Missing brand.')
      setLoading(false)
      return
    }

    setLoading(true)
    setError('')
    setMessage('')

    try {
      const query = new URLSearchParams({ brand })

      if (selectedProductId) {
        query.set('productId', selectedProductId)
      }

      const response = await fetch(
        `/api/artworks/${id}?${query.toString()}`,
        { cache: 'no-store' }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to load artwork.')
      }

      setArtwork(result)

      const metadata = result.metadata ?? defaultProductMetadata

      const loadedDetails: ProductMetadata = {
        ...defaultProductMetadata,
        ...metadata,
        slug: metadata.slug || slugify(metadata.title)
      }

      setSlugManuallyEdited(
        Boolean(metadata.slug && metadata.slug !== slugify(metadata.title))
      )

      setDetails(loadedDetails)
      setPriceInput(
        loadedDetails.price === 0 ? '' : String(loadedDetails.price)
      )
      setSavedDetails(loadedDetails)
      setSlugAvailable(null)

    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Something went wrong.'
      )
    } finally {
      setLoading(false)
    }
  }

  loadArtwork()
}, [id, brand, selectedProductId])

  useEffect(() => {
    if (!artwork) return

    let cancelled = false

    const product = productCatalog.find(
      (item) => item.id === details.productType
    )

    if (!product) {
      setAvailableMockups([])
      setCheckingMockups(false)
      return
    }

    const colors = product.colors.filter(
      (color) => details.colors.includes(color.id)
    )

    async function checkMockups() {
      setCheckingMockups(true)

      const results = await Promise.all(
        colors.map(async (color) => {
          const url = new URL(
            `/api/mockups/${artwork!.id}`,
            window.location.origin
          )

          url.searchParams.set('brand', artwork!.brand)
          url.searchParams.set('productType', product!.id)
          url.searchParams.set('color', color.id)

          try {
            const response = await fetch(url.toString(), {
              method: 'HEAD',
              cache: 'no-store',
            })

            return response.ok ? color.id : null
          } catch {
            return null
          }
        })
      )

      if (!cancelled) {
        setAvailableMockups(
          results.filter((color): color is string => color !== null)
        )
        setCheckingMockups(false)
      }
    }

    checkMockups()

    return () => {
      cancelled = true
    }
  }, [artwork, details.productType, details.colors])


  useEffect(() => {
    if (!artwork) return

    const slug = details.slug

    if (!slug) {
      setSlugAvailable(null)
      setCheckingSlug(false)
      return
    }

    let cancelled = false

    setCheckingSlug(true)
    setSlugAvailable(null)

    const timeout = setTimeout(async () => {
      try {
        const params = new URLSearchParams({
          brand: artwork.brand,
          slug,
          artworkId: artwork.id,
        })

        const response = await fetch(
          `/api/slugs/check?${params.toString()}`,
          { cache: 'no-store' }
        )

        if (!response.ok) {
          throw new Error('Unable to check slug.')
        }

        const result: { available: boolean } = await response.json()

        if (!cancelled) {
          setSlugAvailable(result.available)
        }
      } catch {
        if (!cancelled) {
          setSlugAvailable(null)
        }
      } finally {
        if (!cancelled) {
          setCheckingSlug(false)
        }
      }
    }, 400)

    return () => {
      cancelled = true
      clearTimeout(timeout)
    }
  }, [artwork, details.slug])

  async function handleGenerateMockup() {
    if (!artwork || generating || checkingMockups) return

    const product = productCatalog.find(
      (item) => item.id === details.productType
    )

    if (
      !product ||
      !details.colors.includes(selectedColor)
    ) {
      setError('Select a valid product color first.')
      return
    }

    setGenerating(true)
    setMessage('')
    setError('')

    try {
      const response = await fetch('/api/mockups', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          brand: artwork.brand,
          id: artwork.id,
          productType: product.id,
          color: selectedColor,
        }),
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(
          result.error || 'Unable to generate mockup.'
        )
      }

      setAvailableMockups((current) =>
        current.includes(selectedColor)
          ? current
          : [...current, selectedColor]
      )

      setMockupVersion(Date.now())
      setMessage('Mockup generated successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to generate mockup.'
      )
    } finally {
      setGenerating(false)
    }
  }

  if (loading) {
    return (
      <main className="p-12">
        Loading artwork...
      </main>
    )
  }

  if (error && !artwork) {
    return (
      <main className="p-12">
        <p className="text-red-700">{error}</p>
        <Link href="/library" className="mt-4 inline-block underline">
          Back to library
        </Link>
      </main>
    )
  }

  if (!artwork) return null

  const brandName = brands.find((item) => item.id === artwork.brand)?.name ?? artwork.brand

  async function handleSaveDetails() {
    if (!artwork || saving) return

    setSaving(true)
    setMessage('')
    setError('')

    try {
      const query = new URLSearchParams({
        brand: artwork.brand
      })

      if (artwork.productId) {
        query.set('productId', artwork.productId)
      }

      const response = await fetch(
        `/api/artworks/${artwork.id}?${query.toString()}`,
        {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(details)
        }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(
          result.error || 'Unable to save product details.'
        )
      }

      const artworkResponse = await fetch(
        `/api/artworks/${artwork.id}?${query.toString()}`,
        { cache: 'no-store' }
      )

      if (!artworkResponse.ok) {
        throw new Error('Product saved, but unable to refresh product data.')
      }

      const updatedArtwork: ArtworkDetail = await artworkResponse.json()

      setArtwork(updatedArtwork)
      setSavedDetails(details)
      setMessage('Product details saved successfully!')

    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to save product details.'
      )
    } finally {
      setSaving(false)
    }
  }

  const hasUnsavedChanges = savedDetails !== null && JSON.stringify(details) !== JSON.stringify(savedDetails)

  async function handleDeleteProduct() {
    if (!artwork?.productId || deletingProduct) return

    const confirmed = window.confirm(
      'Permanently delete this product?\n\n' +
      'It will also be removed from the storefront.\n' +
      'The original artwork and mockups will be preserved.'
    )

    if (!confirmed) return

    setDeletingProduct(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(
        `/api/products/${artwork.productId}`,
        { method: 'DELETE' }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(
          result.error || 'Unable to delete product.'
        )
      }

      const resetDetails = { ...defaultProductMetadata }

      setArtwork((current) =>
        current
          ? {
              ...current,
              productId: null,
              isPublished: false,
              metadata: resetDetails
            }
          : current
      )

      setDetails(resetDetails)
      setSavedDetails(resetDetails)
      setPriceInput('')
      setSlugManuallyEdited(false)
      setSlugAvailable(null)

      setMessage('Product deleted. Original artwork preserved!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to delete product.'
      )
    } finally {
      setDeletingProduct(false)
    }
  }

  async function handleDelete() {
    if (!artwork || deleting) return

    const confirmed = window.confirm(
      'Are you sure you want to permanently delete this artwork and all its mockups? This cannot be undone.'
    )

    if (!confirmed) return

    setDeleting(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(
        `/api/artworks/${artwork.id}?brand=${encodeURIComponent(artwork.brand)}`,
        { method: 'DELETE' }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to delete artwork.')
      }

      router.replace('/library')
      router.refresh()
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to delete artwork.'
      )
      setDeleting(false)
    }
  }

  async function handleExport() {
    if (!artwork || exporting || hasUnsavedChanges) return

    setExporting(true)
    setError('')
    setMessage('')

    try {
      const url = `/api/artworks/${artwork.id}/export?brand=${encodeURIComponent(artwork.brand)}`

      const response = await fetch(url)

      if (!response.ok) {
        const result = await response.json()
        throw new Error(result.error || 'Unable to export product.')
      }

      const blob = await response.blob()
      const downloadUrl = URL.createObjectURL(blob)

      const link = document.createElement('a')
      link.href = downloadUrl
      link.download = `${artwork.brand}-${details.slug}.zip`

      document.body.appendChild(link)
      link.click()
      link.remove()

      window.setTimeout(() => URL.revokeObjectURL(downloadUrl), 1000)

      setMessage('Product ZIP downloaded successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to export product.'
      )
    } finally {
      setExporting(false)
    }
  }

  async function handlePublish() {
    if (
      !artwork?.productId ||
      publishing ||
      hasUnsavedChanges ||
      details.status !== 'ready'
    ) return

    setPublishing(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(
        `/api/products/${artwork.productId}/publish`,
        { method: 'POST' }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to publish product.')
      }

      setArtwork((current) =>
        current ? { ...current, isPublished: true } : current
      )

      setMessage('Product published successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to publish product.'
      )
    } finally {
      setPublishing(false)
    }
  }

  async function handleUnpublish() {
    if (!artwork?.productId || unpublishing) return

    const confirmed = window.confirm(
      'Remove this product from the storefront? Your Studio product and mockups will be preserved.'
    )

    if (!confirmed) return

    setUnpublishing(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(
        `/api/products/${artwork.productId}/publish`,
        { method: 'DELETE' }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to unpublish product.')
      }

      setArtwork((current) =>
        current ? { ...current, isPublished: false } : current
      )

      setMessage('Product unpublished successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to unpublish product.'
      )
    } finally {
      setUnpublishing(false)
    }
  }

  async function handleCreateProduct() {
    if (!artwork || creatingProduct) return

    if (hasUnsavedChanges) {
      setError('Save or discard your changes before creating a new product.')
      return
    }

    setCreatingProduct(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(
        `/api/artworks/${artwork.id}/products?brand=${encodeURIComponent(artwork.brand)}`,
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            productType: details.productType
          })
        }
      )

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to create product.')
      }

      window.location.href =
        `/library/${artwork.id}?brand=${encodeURIComponent(artwork.brand)}&productId=${encodeURIComponent(result.productId)}`
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to create product.'
      )
    } finally {
      setCreatingProduct(false)
    }
  }

  const selectedProduct = productCatalog.find(
    (product) => product.id === details.productType
  )

  const selectedColors = selectedProduct?.colors.filter(
    (color) => details.colors.includes(color.id)
  ) ?? []

  const activeColor = selectedColors.find(
    (color) => color.id === selectedColor
  ) ?? selectedColors[0]

  const activeColorId = activeColor?.id

  const hasActiveMockup = activeColorId
    ? availableMockups.includes(activeColorId)
    : false

  return (
    <main className="min-h-screen bg-[#F5F2EB] px-6 py-12 text-[#171717]">
      <div className="mx-auto max-w-6xl">
        <Link
          href="/library"
          className="mb-8 inline-block text-sm font-medium text-neutral-600 hover:text-black"
        >
          ← Back to library
        </Link>

        <header className="mb-10 flex flex-wrap items-start justify-between gap-6">
          <div>
            <p className="mb-2 text-sm uppercase tracking-widest text-neutral-500">
              {brandName}
            </p>

            <h1 className="text-4xl font-bold">
              Artwork Details
            </h1>

            <p className="mt-3 text-neutral-600">
              Uploaded {new Date(artwork.createdAt).toLocaleDateString()}
            </p>
          </div>

          <button
            type="button"
            onClick={handleDelete}
            disabled={deleting || saving || generating || exporting}
            className="rounded-lg border border-red-200 px-5 py-3 text-sm font-semibold text-red-700 transition hover:border-red-600 hover:bg-red-50 disabled:cursor-not-allowed disabled:opacity-40"
          >
            {deleting ? 'Deleting...' : 'Delete Artwork'}
          </button>
        </header>

        <div className="grid gap-8 md:grid-cols-2">
          <section className="rounded-2xl border border-neutral-200 bg-white p-6">
            <h2 className="mb-4 text-lg font-semibold">
              Original artwork
            </h2>

            <div className="relative aspect-square overflow-hidden rounded-xl bg-neutral-100">
              <Image
                src={`/api/artworks/${artwork.id}/image?brand=${artwork.brand}`}
                alt="Original artwork"
                fill
                unoptimized
                sizes="(max-width: 768px) 100vw, 50vw"
                className="object-contain p-6"
              />
            </div>
          </section>

          {/* Product Image Gallery */}
          <section className="rounded-2xl border border-neutral-200 bg-white p-6">
            <div className="mb-5">
              <h2 className="text-lg font-semibold">
                Product Image Gallery
              </h2>

              <p className="mt-1 text-sm text-neutral-500">
                Preview and generate mockups for each selected color.
              </p>
            </div>

            {/* Main preview */}
            <div className="relative aspect-square overflow-hidden rounded-xl bg-[#F5F2EB]">
              {checkingMockups ? (
                <div className="flex h-full items-center justify-center text-sm text-neutral-500">
                  Checking mockups...
                </div>
              ) : hasActiveMockup && activeColorId ? (
                <Image
                  key={`${details.productType}-${activeColorId}-${mockupVersion}`}
                  src={`/api/mockups/${artwork.id}?brand=${artwork.brand}&productType=${details.productType}&color=${activeColorId}&v=${mockupVersion}`}
                  alt={`${activeColor?.name} ${selectedProduct?.name} mockup`}
                  fill
                  unoptimized
                  sizes="(max-width: 768px) 100vw, 50vw"
                  className="object-contain p-6"
                />
              ) : (
                <div className="flex h-full flex-col items-center justify-center gap-3 text-neutral-500">
                  <Shirt size={36} strokeWidth={1.5} className="text-neutral-400" />
                  <p className="text-sm">
                    No mockup generated for this color yet.
                  </p>
                </div>
              )}
            </div>

            {/* Color thumbnails */}
            <div className="mt-5">
              <p className="mb-3 text-sm font-medium">
                Product colors
              </p>

              <div className="flex flex-wrap gap-3">
                {selectedColors.map((color) => {
                  const isActive = activeColorId === color.id
                  const hasImage = availableMockups.includes(color.id)

                  return (
                    <button
                      key={color.id}
                      type="button"
                      onClick={() => {
                        setSelectedColor(color.id)
                        setMessage('')
                        setError('')
                      }}
                      disabled={generating}
                      aria-pressed={isActive}
                      className={`flex min-w-20 flex-col items-center gap-2 rounded-xl border p-3 transition ${
                        isActive
                          ? 'border-neutral-900 bg-neutral-100'
                          : 'border-neutral-200 hover:border-neutral-400'
                      } disabled:opacity-50`}
                    >
                      <span
                        className="h-9 w-9 rounded-full border border-neutral-300"
                        style={{ backgroundColor: color.hex }}
                      />

                      <span className="text-xs font-medium">
                        {color.name}
                      </span>

                      <span className={`text-[10px] ${
                        hasImage ? 'text-green-700' : 'text-neutral-400'
                      }`}>
                        {hasImage ? 'Ready' : 'Not generated'}
                      </span>
                    </button>
                  )
                })}
              </div>
            </div>

            {/* Generate button */}
            <button
              type="button"
              onClick={handleGenerateMockup}
              disabled={generating || checkingMockups || !activeColorId}
              className="mt-6 w-full rounded-lg bg-neutral-900 px-5 py-3 font-semibold text-white transition hover:bg-neutral-700 disabled:opacity-40"
            >
              {generating
                ? 'Generating...'
                : hasActiveMockup
                  ? `Regenerate ${activeColor?.name} mockup`
                  : `Generate ${activeColor?.name ?? ''} mockup`}
            </button>

            {message && (
              <p className="mt-3 text-sm text-green-700">
                {message}
              </p>
            )}

            {error && (
              <p className="mt-3 text-sm text-red-700">
                {error}
              </p>
            )}
          </section>
        </div>

        {/* Product Options */}
        <section className="mt-8 rounded-2xl border border-neutral-200 bg-white p-6">
          <div className="mb-6">
            <h2 className="text-xl font-semibold">
              Product Options
            </h2>

            <p className="mt-1 text-sm text-neutral-500">
              Choose which products and colors this artwork will be available on.
            </p>
          </div>

          {/* Product type */}
          <div className="mb-8">
            <label
              htmlFor="product-type"
              className="mb-2 block text-sm font-medium"
            >
              Product type
            </label>

            <div className="relative w-full md:max-w-sm">
              <select
                id="product-type"
                value={details.productType}
                onChange={(event) => {
                  const selected = productCatalog.find(
                    (product) => product.id === event.target.value
                  )

                  if (!selected) return

                  setDetails((current) => ({
                    ...current,
                    productType: selected.id,
                    colors: selected.colors.length
                      ? [selected.colors[0].id]
                      : [],
                    sizes: current.sizes.filter((size) =>
                      selected.sizes.includes(size)
                    )
                  }))
                }}
                className="w-full appearance-none cursor-pointer rounded-lg border border-neutral-300 bg-white py-3 pl-5 pr-12 text-sm text-neutral-700 transition hover:border-neutral-400 focus:border-neutral-500 focus:outline-none"
              >
                {productCatalog.map((product) => (
                  <option key={product.id} value={product.id}>
                    {product.name}
                  </option>
                ))}
              </select>

              <ChevronDown
                size={18}
                strokeWidth={1.75}
                aria-hidden="true"
                className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-neutral-500"
              />
            </div>
          </div>

          {/* Colors */}
          {productCatalog
            .filter((product) => product.id === details.productType)
            .map((product) => (
              <div key={product.id}>
                <p className="mb-3 text-sm font-medium">
                  Available colors
                </p>

                <div className="flex flex-wrap gap-4">
                  {product.colors.map((color) => {
                    const selected = details.colors.includes(color.id)

                    return (
                      <button
                        key={color.id}
                        type="button"
                        onClick={() => {
                          setDetails((current) => ({
                            ...current,
                            colors: selected
                              ? current.colors.length > 1
                                ? current.colors.filter(
                                    (id) => id !== color.id
                                  )
                                : current.colors
                              : [...current.colors, color.id],
                          }))
                        }}
                        aria-pressed={selected}
                        className="flex flex-col items-center gap-2"
                      >
                        <span
                          className={`flex h-12 w-12 items-center justify-center rounded-full border-2 transition ${
                            selected
                              ? 'border-neutral-900 ring-2 ring-neutral-300 ring-offset-2'
                              : 'border-neutral-300 hover:border-neutral-600'
                          }`}
                          style={{ backgroundColor: color.hex }}
                        >
                          {selected && (
                            <span className="text-lg font-bold text-white">
                              ✓
                            </span>
                          )}
                        </span>

                        <span className="text-sm text-neutral-600">
                          {color.name}
                        </span>
                      </button>
                    )
                  })}
                </div>

                <p className="mt-4 text-sm text-neutral-500">
                  {details.colors.length} color(s) selected
                </p>
              </div>
            ))}
        </section>

        {/* Product Detail */}
        <section className="mt-8 rounded-2xl border border-neutral-200 bg-white p-6">
          <div className="mb-6">
            <h2 className="text-xl font-semibold">
              Product Details
            </h2>
            <p className="mt-1 text-sm text-neutral-500">
              Prepare this artwork for your storefront.
            </p>
          </div>

          <div className="grid gap-6 md:grid-cols-2">
            <div className="md:col-span-2">
              <label
                htmlFor="product-title"
                className="mb-2 block text-sm font-medium"
              >
                Product name
              </label>

              <input
                id="product-title"
                type="text"
                maxLength={150}
                value={details.title}
                onChange={(event) => {
                  const title = event.target.value

                  setDetails((current) => ({
                    ...current,
                    title,
                    slug: slugManuallyEdited
                      ? current.slug
                      : slugify(title)
                  }))

                placeholder="Medusa: Making Men Hard Since 700 BC"
                className="w-full rounded-lg border border-neutral-300 px-4 py-3"

                          ? 'text-red-700'

                      ? 'Slug is available'

                Available sizes

                htmlFor="product-category"

              >
                Category
              </label>

              <div className="relative w-full md:max-w-sm">

                  onChange={(event) =>

                      category: event.target.value,

                >
                  <option value="">Select a category</option>

                  {productOptions[artwork.brand].categories.map(
                    (category) => (
                      <option key={category.id} value={category.id}>
                        {category.name}
                      </option>
                    )

                </select>

                <ChevronDown
                  size={18}
                  strokeWidth={1.75}
                  aria-hidden="true"
                  className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-neutral-500"

                className="mb-2 block text-sm font-medium"

                <select
                  id="product-label"
                  value={details.label}
                  onChange={(event) =>
                    setDetails((current) => ({
                      ...current,
                      label: event.target.value
                    }))
                  }
                  className="w-full appearance-none cursor-pointer rounded-lg border border-neutral-300 bg-white py-3 pl-5 pr-12 text-sm text-neutral-700 transition hover:border-neutral-400 focus:border-neutral-500 focus:outline-none"

                  strokeWidth={1.75}
                  aria-hidden="true"
                  className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-neutral-500"
                />
              </div>
            </div>

            {/* Status */}
            <div>

                htmlFor="product-status"
                className="mb-2 block text-sm font-medium"
              >
                Status
              </label>

              <div className="relative w-full md:max-w-sm">
                <select
                  id="product-status"
                  value={details.status}
                  onChange={(event) =>
                    setDetails((current) => ({
                      ...current,
                      status: event.target.value as ProductMetadata['status'],
                    }))
                  }
                  className="w-full appearance-none cursor-pointer rounded-lg border border-neutral-300 bg-white py-3 pl-5 pr-12 text-sm text-neutral-700 transition hover:border-neutral-400 focus:border-neutral-500 focus:outline-none"

                  <option value="draft">Draft</option>
                  <option value="ready">Ready for export</option>
                </select>

                  aria-hidden="true"

                />
              </div>
            </div>

            <div className="md:col-span-2">
              <p className="mb-3 text-sm font-medium">
                Collections
              </p>

              <div className="flex flex-wrap gap-4">
                {productOptions[artwork.brand].collections.map(
                  (collection) => (
                    <label
                      key={collection.id}
                      className="flex cursor-pointer items-center gap-2 text-sm"
                    >
                      <input
                        type="checkbox"
                        checked={details.collections.includes(collection.id)}
                        onChange={(event) => {
                          setDetails((current) => ({
                            ...current,
                            collections: event.target.checked
                              ? [...current.collections, collection.id]
                              : current.collections.filter(
                                  (id) => id !== collection.id
                                )
                          }))
                        }}
                      />

                      {collection.name}
                    </label>
                  )
                )}
              </div>
            </div>
          </div>

          <div className="mt-8 flex flex-wrap gap-3">
            <button
              type="button"
              onClick={handleSaveDetails}
              disabled={
                saving ||
                checkingSlug ||
                (details.slug !== '' && slugAvailable !== true)
              }
              className="rounded-lg bg-neutral-900 px-6 py-3 font-semibold text-white transition hover:bg-neutral-700 disabled:opacity-40"
            >
              {saving ? 'Saving...' : 'Save details'}
            </button>

            <button
              type="button"
              onClick={handlePublish}
              disabled={
                publishing ||
                unpublishing ||
                saving ||
                generating ||
                checkingSlug ||
                checkingMockups ||
                hasUnsavedChanges ||
                !artwork.productId ||
                details.status !== 'ready' ||
                !details.colors.every((color) =>
                  availableMockups.includes(color)
                )
              }
              className="rounded-lg bg-emerald-700 px-6 py-3 font-semibold text-white transition hover:bg-emerald-800 disabled:cursor-not-allowed disabled:opacity-40"
            >
              {publishing
                ? 'Publishing...'
                : artwork.isPublished
                  ? 'Update Storefront'
                  : 'Publish to Storefront'}
            </button>

            {artwork.isPublished && (
              <button
                type="button"
                onClick={handleUnpublish}
                disabled={unpublishing || publishing || saving}
                className="rounded-lg border border-red-300 px-6 py-3 font-semibold text-red-700 transition hover:bg-red-50 disabled:opacity-40"
              >
                {unpublishing ? 'Unpublishing...' : 'Unpublish'}
              </button>
            )}

            {artwork.productId && (
              <button
                type="button"
                onClick={handleDeleteProduct}
                disabled={
                  deletingProduct ||
                  saving ||
                  publishing ||
                  unpublishing
                }
                className="rounded-lg border border-red-300 px-6 py-3 font-semibold text-red-700 transition hover:bg-red-50 disabled:cursor-not-allowed disabled:opacity-40"
              >
                {deletingProduct ? 'Deleting product...' : 'Delete Product'}
              </button>
            )}

            <button
              type="button"
              onClick={handleExport}
              disabled={
                exporting ||
                saving ||
                generating ||
                hasUnsavedChanges ||
                details.status !== 'ready' ||
                checkingMockups ||
                !details.colors.every((color) =>
                  availableMockups.includes(color)
                )
              }
              className="rounded-lg border border-neutral-300 px-6 py-3 font-semibold text-neutral-700 transition hover:bg-neutral-100 disabled:opacity-40"
            >
              {exporting ? 'Exporting...' : 'Download ZIP'}
            </button>
          </div>

          {hasUnsavedChanges && (
            <p className="mt-3 text-sm font-medium text-amber-700">
              You have unsaved changes.
            </p>
          )}

          {message && (
            <p className="mt-3 text-sm text-green-700">
              {message}
            </p>
          )}

          {error && (
            <p className="mt-3 text-sm text-red-700">
              {error}
            </p>
          )}
        </section>

        {/* Linked Products */}
        <section className="mt-8 rounded-xl border border-neutral-200 bg-white p-6 shadow-sm">
          <div className="mb-6 flex flex-wrap items-center justify-between gap-3">
            <div>
              <h2 className="text-xl font-bold">Linked Products</h2>
              <p className="mt-1 text-sm text-neutral-500">
                Products using this artwork.
              </p>
            </div>

            <div className="flex items-center gap-3">
              <span className="rounded-full bg-neutral-100 px-3 py-1 text-sm font-medium text-neutral-700">
                {artwork.linkedProducts?.length ?? 0} products
              </span>

              <button
                type="button"
                onClick={handleCreateProduct}
                disabled={creatingProduct || saving || deletingProduct}
                className="rounded-lg bg-neutral-900 px-4 py-2 text-sm font-semibold text-white transition hover:bg-neutral-700 disabled:opacity-40"
              >
                {creatingProduct ? 'Creating...' : '+ Create Product'}
              </button>
            </div>
          </div>

          {artwork.linkedProducts?.length ? (
            <div className="divide-y divide-neutral-200">
              {artwork.linkedProducts.map((product) => (
                <button
                  key={product.id}
                  type="button"
                  onClick={() => {
                    if (hasUnsavedChanges) {
                      const confirmed = window.confirm(
                        'You have unsaved changes. Discard them and open another product?'
                      )

                      if (!confirmed) return
                    }

                    window.location.href =
                      `/library/${artwork.id}?brand=${encodeURIComponent(artwork.brand)}&productId=${encodeURIComponent(product.id)}`
                  }}
                  className={`flex w-full flex-wrap items-center justify-between gap-4 py-4 text-left transition hover:bg-neutral-50 ${
                    product.id === artwork.productId ? 'bg-neutral-50' : ''
                  }`}
                >
                  <div>
                    <h3 className="font-semibold">
                      {product.title || 'Untitled product'}
                    </h3>

                    <p className="mt-1 text-sm text-neutral-500">
                      {product.productType}
                    </p>
                  </div>

                  <div className="flex items-center gap-4">
                    <span
                      className={`rounded-full px-3 py-1 text-xs font-semibold ${
                        product.isPublished
                          ? 'bg-green-100 text-green-800'
                          : product.status === 'ready'
                            ? 'bg-blue-100 text-blue-800'
                            : 'bg-neutral-100 text-neutral-600'
                      }`}
                    >
                      {product.isPublished
                        ? 'Published'
                        : product.status === 'ready'
                          ? 'Ready'
                          : 'Draft'}
                    </span>

                    {product.id === artwork.productId && (
                      <span className="text-xs font-medium text-neutral-400">
                        Current product
                      </span>
                    )}
                  </div>
                </button>
              ))}
            </div>
          ) : (
            <p className="text-sm text-neutral-500">
              No products linked to this artwork yet.
            </p>
          )}
        </section>
      </div>
    </main>
  )
}

'use client'

import { useEffect, useState } from 'react'
import Image from 'next/image'
import { brands, type BrandId } from '@/lib/brands'
import Link from 'next/link'
import { ChevronDown } from 'lucide-react'

export default function HomePage() {
  const [file, setFile] = useState<File | null>(null)
  const [preview, setPreview] = useState<string | null>(null)
  const [brand, setBrand] = useState<BrandId>('legends')
  const [uploading, setUploading] = useState(false)
  const [message, setMessage] = useState('')
  const [artworkId, setArtworkId] = useState<string | null>(null)
  const [generating, setGenerating] = useState(false)
  const [mockupUrl, setMockupUrl] = useState<string | null>(null)

  useEffect(() => {
    if (!file) {
      setPreview(null)
      return
    }

    const url = URL.createObjectURL(file)
    setPreview(url)

    return () => URL.revokeObjectURL(url)
  }, [file])

  function handleFileChange(
    event: React.ChangeEvent<HTMLInputElement>
  ) {
    const selectedFile = event.target.files?.[0]

    if (!selectedFile) return

    if (selectedFile.type !== 'image/png') {
      alert('Please upload a PNG file.')
      return
    }
    setArtworkId(null)
    setMockupUrl(null)
    setMessage('')
    setFile(selectedFile)
  }

  async function handleUpload() {
    if (!file || uploading || generating) return

    setUploading(true)
    setMessage('')
    setArtworkId(null)
    setMockupUrl(null)

    try {
      const formData = new FormData()
      formData.append('artwork', file)
      formData.append('brand', brand)

      const response = await fetch('/api/artworks', {
        method: 'POST',
        body: formData
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Upload failed.')
      }

      setArtworkId(result.id)
      setMessage(`Artwork saved successfully! (${result.width} × ${result.height})`)

    } catch (error) {
      setMessage(error instanceof Error ? error.message : 'Upload failed.')
    } finally {
      setUploading(false)
    }
  }

  async function handleGenerateMockup() {
    if (!artworkId || generating) return

    setGenerating(true)
    setMessage('')

    try {
      const response = await fetch('/api/mockups', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          brand,
          id: artworkId,
        }),
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(
          result.error || 'Mockup generation failed.'
        )
      }

      setMockupUrl(`/api/mockups/${artworkId}?brand=${brand}&v=${Date.now()}`)
      setMessage('Mockup generated successfully!')

    } catch (error) {
      setMessage(
        error instanceof Error
          ? error.message
          : 'Mockup generation failed.'
      )
    } finally {
      setGenerating(false)
    }
  }

  return (
    <main className="min-h-screen bg-[#F5F2EB] px-6 py-12 text-[#171717]">
      <div className="mx-auto max-w-7xl">
        <header className="mb-10 flex flex-wrap items-end justify-between gap-6">
          <div>
            <p className="mb-2 text-sm font-semibold uppercase tracking-[0.3em] text-neutral-500">
              NVT Company
            </p>

            <h1 className="text-4xl font-bold tracking-tight">
              Design Studio
            </h1>
          </div>
          <div className="flex flex-wrap items-center gap-3">
            <Link
              href="/library"
              className="rounded-lg bg-neutral-900 px-5 py-3 text-sm font-semibold text-white transition hover:bg-neutral-700"
            >
              View artwork library →
            </Link>
          </div>
        </header>
        <div className="grid gap-8 md:grid-cols-2">
          <section className="rounded-2xl border border-neutral-200 bg-white p-8">
            <h2 className="mb-6 text-xl font-semibold">
              Upload artwork
            </h2>
            <label className="flex min-h-64 cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed border-neutral-300 bg-neutral-50 p-6 text-center transition hover:border-neutral-500">
              <span className="mb-3 text-4xl">↑</span>
              <span className="font-medium">
                Choose your artwork
              </span>
              <span className="mt-2 text-sm text-neutral-500">
                Transparent background and PNG recommended
              </span>

              <input
                type="file"
                accept="image/png"
                disabled={uploading || generating}
                onChange={handleFileChange}
                className="sr-only"
              />
            </label>

            {file && (
              <div className="mt-5 text-sm">
                <p className="font-medium">{file.name}</p>
                <p className="text-neutral-500">
                  {(file.size / 1024 / 1024).toFixed(2)} MB
                </p>
              </div>
            )}
            <div className="mt-6 space-y-4">
              <div>
                <label
                  htmlFor="brand"
                  className="mb-2 block text-sm font-medium"
                >
                  Brand
                </label>

                <div className="relative">
                  <select
                    id="brand"
                    value={brand}
                    disabled={uploading || generating}
                    onChange={(event) => {
                      setBrand(event.target.value as BrandId)
                      setArtworkId(null)
                      setMockupUrl(null)
                      setMessage('')
                    }}
                    className="w-full appearance-none cursor-pointer rounded-lg border border-neutral-300 bg-white py-3 pl-5 pr-12 text-sm text-neutral-700 transition hover:border-neutral-400 focus:border-neutral-500 focus:outline-none disabled:cursor-not-allowed disabled:opacity-50"
                  >
                    {brands.map((item) => (
                      <option key={item.id} value={item.id}>
                        {item.name}
                      </option>
                    ))}
                  </select>

                  <ChevronDown
                    size={18}
                    strokeWidth={1.75}
                    aria-hidden="true"
                    className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-neutral-500"
                  />
                </div>
              </div>

              <button
                type="button"
                onClick={handleUpload}
                disabled={!file || uploading || generating}
                className="w-full rounded-lg bg-neutral-900 px-5 py-3 font-semibold text-white transition hover:bg-neutral-700 disabled:cursor-not-allowed disabled:opacity-40"
              >
                {uploading ? 'Uploading...' : 'Save artwork'}
              </button>

              <button
                type="button"
                onClick={handleGenerateMockup}
                disabled={!artworkId || generating || uploading}
                className="w-full rounded-lg bg-[#171717] px-5 py-3 font-semibold text-white transition hover:bg-neutral-700 disabled:cursor-not-allowed disabled:opacity-40"
              >
                {generating ? 'Generating...' : 'Generate mockup'}
              </button>

              {message && (
                <p role="status" className="text-sm text-neutral-600">
                  {message}
                </p>
              )}
            </div>
          </section>
          <section className="rounded-2xl border border-neutral-200 bg-white p-8">
            <h2 className="mb-6 text-xl font-semibold">
              Preview
            </h2>

            <div className="grid gap-6">
              <div>
                <h3 className="mb-3 text-sm font-medium">
                  Original artwork
                </h3>

                <div className="flex min-h-64 items-center justify-center overflow-hidden rounded-xl bg-neutral-100 p-6">
                  {preview ? (
                    <Image
                      src={preview}
                      alt="Artwork preview"
                      width={400}
                      height={400}
                      unoptimized
                      className="max-h-64 w-auto object-contain"
                    />
                  ) : (
                    <p className="text-sm text-neutral-500">
                      Your artwork will appear here.
                    </p>
                  )}
                </div>
              </div>

              <div>
                <h3 className="mb-3 text-sm font-medium">
                  Generated mockup
                </h3>

                <div className="flex min-h-80 items-center justify-center overflow-hidden rounded-xl bg-[#F5F2EB] p-6">
                  {mockupUrl ? (
                    <Image
                      src={mockupUrl}
                      alt="Generated shirt mockup"
                      width={500}
                      height={600}
                      unoptimized
                      className="h-auto max-h-96 w-auto object-contain"
                    />
                  ) : (
                    <p className="text-sm text-neutral-500">
                      Generate a mockup to see the result.
                    </p>
                  )}
                </div>
              </div>
            </div>
          </section>

        </div>

      </div>
    </main>
  )
}

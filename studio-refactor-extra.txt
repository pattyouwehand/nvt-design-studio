  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }
  try {
    const body = await request.json()

    const {
      brand,
      id,
      productType = 'classic-tee',
      color = 'black',
    } = body

    if (
      typeof brand !== 'string' ||
      !isBrandId(brand) ||
      typeof id !== 'string' ||

      : undefined

    ) {
      return NextResponse.json(
        { error: 'Invalid product type or color.' },
        { status: 400 }
      )

      brand,
      id,
      productType,
      color
    )

    return NextResponse.json({
      success: true,
      filename: result.filename,

========== app/api/artworks/[id]/export/route.ts ==========

import { NextRequest, NextResponse } from 'next/server'
import JSZip from 'jszip'
import { isBrandId } from '@/lib/brands'
import { isArtworkId } from '@/lib/artworks'
import { validateProductMetadata } from '@/lib/productMetadata'
import { getProductType } from '@/lib/catalog/products'
import { getArtworkById, isArtworkSlugTaken } from '@/lib/artworkDatabase'
import { getArtworkMockup } from '@/lib/artworkStorage'
import { getStudioUser } from '@/lib/auth'

export const runtime = 'nodejs'

export async function GET(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params
  const brand = request.nextUrl.searchParams.get('brand')
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  if (!brand || !isBrandId(brand) || !isArtworkId(id)) {
    return NextResponse.json(
      { error: 'Invalid artwork.' },
      { status: 400 }
    )
  }

  try {
    const artwork = await getArtworkById(brand, id)

    if (!artwork) {
      return NextResponse.json(
        { error: 'Artwork not found.' },
        { status: 404 }
      )
    }

    const metadata = artwork.metadata

    if (!validateProductMetadata(brand, metadata)) {
      return NextResponse.json(
        { error: 'Product details are invalid.' },
        { status: 400 }
      )
    }

    if (metadata.status !== 'ready') {
      return NextResponse.json(
        {
          error: 'Set the product status to Ready for export first.'
        },
        { status: 400 }
      )
    }

    if (await isArtworkSlugTaken(brand, metadata.slug, id)) {
      return NextResponse.json(
        {
          error: 'This URL slug is already in use for this brand.'
        },
        { status: 409 }
      )
    }

    if (
      !metadata.title.trim() ||
      !metadata.slug.trim() ||
      metadata.price <= 0 ||
      metadata.sizes.length === 0 ||
      metadata.colors.length === 0
    ) {
      return NextResponse.json(
        {
          error: 'Complete the product details before exporting.'
        },
        { status: 400 }
      )
    }

    const product = getProductType(metadata.productType)

    if (!product) {
      return NextResponse.json(
        { error: 'Unknown product type.' },
        { status: 400 }
      )
    }

    const zip = new JSZip()
    const images: Record<string, string> = {}

    // Download mockups from Supabase Storage.
    for (const color of metadata.colors) {
      const filename = `${product.id}-${color}-front.webp`

      const image = await getArtworkMockup(
        brand,
        id,
        filename
      )

      const zipPath = `images/${filename}`

      zip.file(zipPath, image)
      images[color] = zipPath
    }

    // Product data for the storefront.
    const exportData = {
      version: 1,
      source: 'nvt-design-studio',
      artworkId: id,
      brand,
      title: metadata.title,
      slug: metadata.slug,
      description: metadata.description,
      price: metadata.price,
      productType: metadata.productType,
      colors: metadata.colors,
      sizes: metadata.sizes,
      category: metadata.category,
      collections: metadata.collections,
      status: 'draft',
      images
    }

    zip.file(
      'product.json',
      JSON.stringify(exportData, null, 2)
    )

    const zipBuffer = await zip.generateAsync({
      type: 'nodebuffer',
      compression: 'DEFLATE',
      compressionOptions: { level: 6 }
    })

    return new NextResponse(new Uint8Array(zipBuffer), {
      headers: {
        'Content-Type': 'application/zip',
        'Content-Disposition':
          `attachment; filename="${brand}-${metadata.slug}.zip"`,
        'Cache-Control': 'no-store'
      }
    })
  } catch (error) {
    console.error('Product export failed:', error)

    return NextResponse.json(
      {
        error:
          'Unable to export product. Check that all selected colors have generated mockups.'
      },
      { status: 500 }
    )
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { randomUUID } from 'node:crypto'
import sharp from 'sharp'
import { getStudioUser } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { supabaseAdmin } from '@/lib/supabaseAdmin'

export const runtime = 'nodejs'

const BUCKET = 'artworks'
const MAX_FILE_SIZE = 25 * 1024 * 1024
const MAX_DIMENSION = 10000

export async function POST(request: NextRequest) {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const formData = await request.formData()

    const file = formData.get('image')
    const name = formData.get('name')
    const productType = formData.get('productType')
    const color = formData.get('color')

    if (!(file instanceof File)) {
      return NextResponse.json(
        { error: 'Please select an image.' },
        { status: 400 }
      )
    }

    if (
      typeof name !== 'string' ||
      !name.trim() ||
      name.trim().length > 150
    ) {
      return NextResponse.json(
        { error: 'Please enter a valid image name.' },
        { status: 400 }
      )
    }

    if (file.size === 0 || file.size > MAX_FILE_SIZE) {
      return NextResponse.json(
        { error: 'Image must be between 1 byte and 25 MB.' },
        { status: 400 }
      )
    }

    if (
      (productType !== null && typeof productType !== 'string') ||
      (color !== null && typeof color !== 'string') ||
      (typeof productType === 'string' && productType.length > 100) ||
      (typeof color === 'string' && color.length > 100)
    ) {
      return NextResponse.json(
        { error: 'Invalid product type or color.' },
        { status: 400 }
      )
    }

    const buffer = Buffer.from(await file.arrayBuffer())
    const metadata = await sharp(buffer).metadata()

    if (
      !metadata.format ||
      !['png', 'jpeg', 'webp'].includes(metadata.format) ||
      !metadata.width ||
      !metadata.height ||
      metadata.width > MAX_DIMENSION ||
      metadata.height > MAX_DIMENSION
    ) {
      return NextResponse.json(
        { error: 'Only valid PNG, JPEG and WebP images are supported.' },
        { status: 400 }
      )
    }

    // Convert the uploaded image to an optimized WebP.
    const optimizedImage = await sharp(buffer)
      .rotate()
      .resize({
        width: 1600,
        height: 1600,
        fit: 'inside',
        withoutEnlargement: true
      })
      .webp({
        quality: 85,
        effort: 6
      })
      .toBuffer()

    const id = randomUUID()
    const storagePath = `shared-images/${id}.webp`

    // Upload the optimized image to Supabase Storage.
    const { error: uploadError } = await supabaseAdmin.storage
      .from(BUCKET)
      .upload(storagePath, optimizedImage, {
        contentType: 'image/webp',
        upsert: false
      })

    if (uploadError) {
      throw uploadError
    }

    try {
      // Save image metadata in PostgreSQL.
      const image = await prisma.sharedImage.create({
        data: {
          id,
          name: name.trim(),
          storagePath,
          productType:
            typeof productType === 'string'
              ? productType.trim() || null
              : null,
          color:
            typeof color === 'string'
              ? color.trim() || null
              : null
        }
      })

      return NextResponse.json(
        {
          success: true,
          image
        },
        { status: 201 }
      )
    } catch (databaseError) {
      // Remove the uploaded file if saving to the database fails.
      const { error: cleanupError } = await supabaseAdmin.storage
        .from(BUCKET)
        .remove([storagePath])

      if (cleanupError) {
        console.error(
          'Unable to clean up shared image:',
          cleanupError
        )
      }

      throw databaseError
    }
  } catch (error) {
    console.error('Shared image upload failed:', error)

    return NextResponse.json(
      { error: 'Unable to upload shared image.' },
      { status: 500 }
    )
  }
}

export async function GET() {
  const user = await getStudioUser()

  if (!user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  try {
    const images = await prisma.sharedImage.findMany({
      orderBy: {
        createdAt: 'desc'
      }
    })

    return NextResponse.json({ images })
  } catch (error) {
    console.error('Unable to load shared images:', error)

    return NextResponse.json(
      { error: 'Unable to load shared images.' },
      { status: 500 }
    )
  }
}

'use client'

import { useEffect, useMemo, useState } from 'react'
import type { ChangeEvent, FormEvent } from 'react'
import Image from 'next/image'
import Link from 'next/link'
import { Pencil, RefreshCw, Trash2 } from 'lucide-react'

interface SharedImage {
  id: string
  name: string
  storagePath: string
  productType: string | null
  color: string | null
  createdAt: string
}

interface EditForm {
  name: string
  productType: string
  color: string
}

const MAX_FILE_SIZE = 15 * 1024 * 1024
const ALLOWED_TYPES = ['image/png', 'image/jpeg', 'image/webp']

export default function SharedImageLibrary() {
  const [images, setImages] = useState<SharedImage[]>([])
  const [loading, setLoading] = useState(true)
  const [uploading, setUploading] = useState(false)
  const [saving, setSaving] = useState(false)
  const [replacing, setReplacing] = useState(false)
  const [error, setError] = useState('')
  const [message, setMessage] = useState('')

  const [file, setFile] = useState<File | null>(null)
  const [name, setName] = useState('')
  const [productType, setProductType] = useState('')
  const [color, setColor] = useState('')
  const [filter, setFilter] = useState('all')

  const [editingId, setEditingId] = useState<string | null>(null)
  const [editForm, setEditForm] = useState<EditForm>({
    name: '',
    productType: '',
    color: ''
  })

  const [replacingId, setReplacingId] = useState<string | null>(null)
  const [replacementFile, setReplacementFile] = useState<File | null>(null)
  const [replacementPreview, setReplacementPreview] = useState<string | null>(null)
  const [deletingId, setDeletingId] = useState<string | null>(null)

  useEffect(() => {
    async function loadImages() {
      try {
        const response = await fetch('/api/shared-images', {
          cache: 'no-store'
        })

        const result = await response.json()

        if (!response.ok) {
          throw new Error(result.error || 'Unable to load images.')
        }

        setImages(result.images)
      } catch (error) {
        setError(
          error instanceof Error
            ? error.message
            : 'Unable to load images.'
        )
      } finally {
        setLoading(false)
      }
    }

    void loadImages()
  }, [])

  useEffect(() => {
    if (!replacementFile) {
      setReplacementPreview(null)
      return
    }

    const previewUrl = URL.createObjectURL(replacementFile)
    setReplacementPreview(previewUrl)

    return () => {
      URL.revokeObjectURL(previewUrl)
    }
  }, [replacementFile])

  const productTypes = useMemo(() => {
    return [...new Set(
      images
        .map((image) => image.productType)
        .filter((value): value is string => Boolean(value))
    )].sort()
  }, [images])

  const filteredImages = useMemo(() => {
    return images.filter(
      (image) =>
        filter === 'all' ||
        image.productType === filter
    )
  }, [images, filter])

  function startEditing(image: SharedImage) {
    if (replacing || saving) return

    cancelReplacing()
    setEditingId(image.id)
    setEditForm({
      name: image.name,
      productType: image.productType ?? '',
      color: image.color ?? ''
    })
    setError('')
    setMessage('')
  }

  function cancelEditing() {
    setEditingId(null)
    setEditForm({
      name: '',
      productType: '',
      color: ''
    })
  }

  async function handleSave(id: string) {
    if (!editForm.name.trim() || saving) return

    setSaving(true)
    setError('')
    setMessage('')

    try {
      const response = await fetch(`/api/shared-images/${id}`, {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          name: editForm.name.trim(),
          productType: editForm.productType.trim() || null,
          color: editForm.color.trim() || null
        })
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to update image.')
      }

      setImages((current) =>
        current.map((image) =>
          image.id === id ? result.image : image
        )
      )

      cancelEditing()
      setMessage('Image updated successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to update image.'
      )
    } finally {
      setSaving(false)
    }
  }

  function startReplacing(image: SharedImage) {
    if (saving || replacing) return

    cancelEditing()
    setReplacingId(image.id)
    setReplacementFile(null)
    setError('')
    setMessage('')
  }

  function cancelReplacing() {
    setReplacingId(null)
    setReplacementFile(null)
  }

  function handleReplacementFile(event: ChangeEvent<HTMLInputElement>) {
    const selectedFile = event.target.files?.[0] ?? null

    if (!selectedFile) {
      setReplacementFile(null)
      return
    }

    if (!ALLOWED_TYPES.includes(selectedFile.type)) {
      setError('Only PNG, JPEG and WebP images are allowed.')
      setReplacementFile(null)
      event.target.value = ''
      return
    }

    if (selectedFile.size === 0 || selectedFile.size > MAX_FILE_SIZE) {
      setError('Image must be smaller than 15 MB.')
      setReplacementFile(null)
      event.target.value = ''
      return
    }

    setError('')
    setMessage('')
    setReplacementFile(selectedFile)
  }

  async function handleReplace(id: string) {
    if (!replacementFile || replacing) return

    setReplacing(true)
    setError('')
    setMessage('')

    try {
      const formData = new FormData()
      formData.append('image', replacementFile)

      const response = await fetch(`/api/shared-images/${id}`, {
        method: 'PUT',
        body: formData
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to replace image.')
      }

      setImages((current) =>
        current.map((image) =>
          image.id === id ? result.image : image
        )
      )

      cancelReplacing()
      setMessage('Image replaced successfully!')
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to replace image.'
      )
    } finally {
      setReplacing(false)
    }
  }

  async function handleDelete(image: SharedImage) {
    const confirmed = window.confirm(
      `Are you sure you want to permanently delete "${image.name}"?\n\nThis action cannot be undone.`
    )

    if (!confirmed) return

    setDeletingId(image.id)
    setError('')
    setMessage('')

    try {
      const response = await fetch(`/api/shared-images/${image.id}`, {
        method: 'DELETE'
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to delete image.')
      }

      setImages((current) =>
        current.filter((item) => item.id !== image.id)
      )

      setMessage(
        result.warning || 'Image deleted successfully!'
      )
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to delete image.'
      )
    } finally {
      setDeletingId(null)
    }
  }

  async function handleUpload(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()

    if (!file || !name.trim()) return

    setUploading(true)
    setError('')
    setMessage('')

    try {
      const formData = new FormData()
      formData.append('image', file)
      formData.append('name', name.trim())
      formData.append('productType', productType.trim())
      formData.append('color', color.trim())

      const response = await fetch('/api/shared-images', {
        method: 'POST',
        body: formData
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Unable to upload image.')
      }

      setImages((current) => [result.image, ...current])

      setFile(null)
      setName('')
      setProductType('')
      setColor('')
      setMessage('Image uploaded successfully!')

      const fileInput = document.getElementById(
        'shared-image-file'
      ) as HTMLInputElement | null

      if (fileInput) fileInput.value = ''
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : 'Unable to upload image.'
      )
    } finally {
      setUploading(false)
    }
  }

  return (
    <main className="min-h-screen bg-[#F5F2EB] px-6 py-12 text-[#171717]">
      <div className="mx-auto max-w-7xl">
        <header className="mb-10 flex flex-wrap items-end justify-between gap-6">
          <div>
            <p className="mb-2 text-sm font-medium uppercase tracking-widest text-neutral-500">
              NVT Design Studio
            </p>

            <h1 className="text-4xl font-bold">
              Shared Image Library
            </h1>

            <p className="mt-3 text-neutral-600">
              Manage reusable product photos for your storefronts.
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-3">
            <Link
              href="/library"
              className="rounded-lg border border-neutral-300 bg-white px-5 py-3 text-sm font-semibold transition hover:bg-neutral-100"
            >
              Artwork Library
            </Link>
          </div>
        </header>

        {/* Overview */}
        <div className="mb-8 grid gap-4 sm:grid-cols-3">
          <div className="rounded-xl border border-neutral-200 bg-white p-5">
            <p className="text-sm text-neutral-500">
              Total images
            </p>
            <p className="mt-2 text-3xl font-bold">
              {images.length}
            </p>
          </div>

          <div className="rounded-xl border border-neutral-200 bg-white p-5">
            <p className="text-sm text-neutral-500">
              Product types
            </p>
            <p className="mt-2 text-3xl font-bold">

                type="file"
                accept="image/png,image/jpeg,image/webp"
                required
                onChange={(event) => {
                  const selectedFile = event.target.files?.[0] ?? null
                  setFile(selectedFile)

                  if (selectedFile && !name) {
                    setName(selectedFile.name.replace(/\.[^.]+$/, ''))
                  }
                }}

            ))}
          </select>
        </div>

        {loading && (
          <p className="text-neutral-500">

                      </div>

                      </p>
                    </>
                  )}
                </div>

        )}
      </div>
    </main>
  )
}

const BUCKET = 'artworks'

export async function uploadArtworkImage( brand: BrandId, id: string, image: Buffer ) {
  const storagePath = getStoragePath(
    brand,
    id,
    'artwork.png'
  )

  const { error } = await supabaseAdmin.storage
    .from(BUCKET)

}

export async function deleteArtworkFiles(brand: BrandId, id: string) {
  const folder = `${brand}/${id}`

  // Find all generated mockups
  const { data: mockups, error: listError } = await supabaseAdmin.storage
    .from(BUCKET)
    .list(`${folder}/mockups`, { limit: 1000 })

  if (listError) throw listError

  const paths = [
    `${folder}/artwork.png`,
    ...(mockups ?? []).map(
      (file) => `${folder}/mockups/${file.name}`
    )
  ]

  const { error } = await supabaseAdmin.storage
    .from(BUCKET)
    .remove(paths)

  if (error) throw error
}

import 'server-only'
import sharp from 'sharp'
import path from 'node:path'
import type { BrandId } from '@/lib/brands'
import { getProductType } from '@/lib/catalog/products'
import {
  getArtworkImage,
  uploadArtworkMockup
} from '@/lib/artworkStorage'

const OUTPUT = {
  width: 1000,
  height: 1200,
  quality: 85
}

export async function generateMockup(
  brand: BrandId,
  id: string,
  productTypeId = 'classic-tee',
  colorId = 'black'
) {
  const productType = getProductType(productTypeId)

  if (!productType) {
    throw new Error(`Unknown product type: ${productTypeId}`)
  }

  const color = productType.colors.find(
    (item) => item.id === colorId
  )

  if (!color) {
    throw new Error(
      `Color "${colorId}" is not available for ${productTypeId}`
    )
  }

  const shirtPath = path.join(
    process.cwd(),
    'assets',
    'mockups',
    color.mockupTemplate
  )

  const filename = `${productTypeId}-${colorId}-front.webp`
  const printArea = productType.printArea

  // Download original artwork from Supabase Storage.
  const originalArtwork = await getArtworkImage(brand, id)

  // Prepare the garment template.
  const shirt = await sharp(shirtPath)
    .resize(OUTPUT.width, OUTPUT.height, {
      fit: 'contain',
      background: {
        r: 0,
        g: 0,
        b: 0,
        alpha: 0
      }
    })
    .ensureAlpha()
    .png()
    .toBuffer()

  // Fit artwork inside the print area.
  const artwork = await sharp(originalArtwork)
    .trim({
      background: '#00000000',
      threshold: 5
    })
    .resize(printArea.width, printArea.height, {
      fit: 'contain',
      background: {
        r: 0,
        g: 0,
        b: 0,
        alpha: 0
      }
    })
    .png()
    .toBuffer()

  // Combine artwork and garment.
  const mockupBuffer = await sharp(shirt)
    .composite([
      {
        input: artwork,
        left: printArea.left,
        top: printArea.top
      }
    ])
    .webp({ quality: OUTPUT.quality })
    .toBuffer()

  // Upload generated mockup to Supabase Storage.
  const storagePath = await uploadArtworkMockup(
    brand,
    id,
    filename,
    mockupBuffer
  )

  return {
    storagePath,
    filename,
    productType: productTypeId,
    color: colorId
  }
}

import 'server-only'
import { createClient } from '@/lib/supabase/server'

export async function getStudioUser() {
  const supabase = await createClient()

  const {
    data: { user },
    error,
  } = await supabase.auth.getUser()

  if (error || !user?.email) {
    return null
  }

  const allowedEmails = (process.env.STUDIO_ALLOWED_EMAILS ?? '')
    .split(',')
    .map((email) => email.trim().toLowerCase())
    .filter(Boolean)

  if (!allowedEmails.includes(user.email.toLowerCase())) {
    return null
  }

  return user
}
